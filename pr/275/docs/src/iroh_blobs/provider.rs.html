42/// Statistics about a successful or failed transfer.
43#[derive(Debug, Serialize, Deserialize)]
44pub struct TransferStats {
45    /// The number of bytes sent that are part of the payload.
46    pub payload_bytes_sent: u64,
47    /// The number of bytes sent that are not part of the payload.
48    ///
49    /// Hash pairs and the initial size header.
50    pub other_bytes_sent: u64,
51    /// The number of bytes read from the stream.
52    ///
53    /// In most cases this is just the request, for push requests this is
54    /// request, size header and hash pairs.
55    pub other_bytes_read: u64,
56    /// Total duration from reading the request to transfer completed.
57    pub duration: Duration,
58}

60/// A pair of [`SendStream`] and [`RecvStream`] with additional context data.
61#[derive(Debug)]
62pub struct StreamPair<R: RecvStream = DefaultReader, W: SendStream = DefaultWriter> {
63    t0: Instant,
64    connection_id: u64,
65    reader: R,
66    writer: W,
67    other_bytes_read: u64,
68    events: EventSender,
69}

71impl StreamPair {
72    pub async fn accept(
73        conn: &endpoint::Connection,
74        events: EventSender,
75    ) -> Result<Self, ConnectionError> {
76        let (writer, reader) = conn.accept_bi().await?;
77        Ok(Self::new(conn.stable_id() as u64, reader, writer, events))
78    }

81impl<R: RecvStream, W: SendStream> StreamPair<R, W> {
82    pub fn stream_id(&self) -> u64 {
83        self.reader.id()
84    }
85
86    pub fn new(connection_id: u64, reader: R, writer: W, events: EventSender) -> Self {
87        Self {
88            t0: Instant::now(),
89            connection_id,
90            reader,
91            writer,
92            other_bytes_read: 0,
93            events,
94        }
95    }
96
97    /// Read the request.
98    ///
99    /// Will fail if there is an error while reading, or if no valid request is sent.
100    ///
101    /// This will read exactly the number of bytes needed for the request, and
102    /// leave the rest of the stream for the caller to read.
103    ///
104    /// It is up to the caller do decide if there should be more data.
105    pub async fn read_request(&mut self) -> Result<Request> {
106        let (res, size) = Request::read_async(&mut self.reader).await?;
107        self.other_bytes_read += size as u64;
108        Ok(res)
109    }
110
111    /// We are done with reading. Return a ProgressWriter that contains the read stats and connection id
112    pub async fn into_writer(
113        mut self,
114        tracker: RequestTracker,
115    ) -> Result<ProgressWriter<W>, io::Error> {
116        self.reader.expect_eof().await?;
117        drop(self.reader);
118        Ok(ProgressWriter::new(
119            self.writer,
120            WriterContext {
121                t0: self.t0,
122                other_bytes_read: self.other_bytes_read,
123                payload_bytes_written: 0,
124                other_bytes_written: 0,
125                tracker,
126            },
127        ))
128    }
129
130    pub async fn into_reader(
131        mut self,
132        tracker: RequestTracker,
133    ) -> Result<ProgressReader<R>, io::Error> {
134        self.writer.sync().await?;
135        drop(self.writer);
136        Ok(ProgressReader {
137            inner: self.reader,
138            context: ReaderContext {
139                t0: self.t0,
140                other_bytes_read: self.other_bytes_read,
141                tracker,
142            },
143        })
144    }
145
146    pub async fn get_request(
147        &self,
148        f: impl FnOnce() -> GetRequest,
149    ) -> Result<RequestTracker, ProgressError> {
150        self.events
151            .request(f, self.connection_id, self.reader.id())
152            .await
153    }
154
155    pub async fn get_many_request(
156        &self,

159        self.events
160            .request(f, self.connection_id, self.reader.id())

162    }

164    pub async fn push_request(
165        &self,

169            .request(f, self.connection_id, self.reader.id())

174        &self,

178            .request(f, self.connection_id, self.reader.id())

182    pub fn stats(&self) -> TransferStats {
183        TransferStats {
184            payload_bytes_sent: 0,
185            other_bytes_sent: 0,
186            other_bytes_read: self.other_bytes_read,
187            duration: self.t0.elapsed(),

285/// Handle a single connection.
286pub async fn handle_connection(
287    connection: endpoint::Connection,
288    store: Store,
289    progress: EventSender,
290) {
291    let connection_id = connection.stable_id() as u64;
292    let span = debug_span!("connection", connection_id);
293    async move {
294        if let Err(cause) = progress
295            .client_connected(|| ClientConnected {
296                connection_id,
297                endpoint_id: Some(connection.remote_id()),
298            })
299            .await
300        {
301            connection.close(cause.code(), cause.reason());
302            debug!("closing connection: {cause}");
303            return;
304        }
305        while let Ok(pair) = StreamPair::accept(&connection, progress.clone()).await {
306            let span = debug_span!("stream", stream_id = %pair.stream_id());
307            let store = store.clone();
308            n0_future::task::spawn(handle_stream(pair, store).instrument(span));
309        }
310        progress
311            .connection_closed(|| ConnectionClosed { connection_id })
312            .await
313            .ok();
314    }
315    .instrument(span)
316    .await
317}

319/// Describes how to handle errors for a stream.
320pub trait ErrorHandler {
321    type W: AsyncStreamWriter;
322    type R: AsyncStreamReader;
323    fn stop(reader: &mut Self::R, code: VarInt) -> impl Future<Output = ()>;
324    fn reset(writer: &mut Self::W, code: VarInt) -> impl Future<Output = ()>;
325}
326
327async fn handle_read_request_result<R: RecvStream, W: SendStream, T, E: HasErrorCode>(
328    pair: &mut StreamPair<R, W>,
329    r: Result<T, E>,
330) -> Result<T, E> {
331    match r {
332        Ok(x) => Ok(x),
333        Err(e) => {
334            pair.writer.reset(e.code()).ok();
335            Err(e)
336        }
337    }
338}
339async fn handle_write_result<W: SendStream, T, E: HasErrorCode>(
340    writer: &mut ProgressWriter<W>,
341    r: Result<T, E>,
342) -> Result<T, E> {
343    match r {
344        Ok(x) => {
345            writer.transfer_completed().await;
346            Ok(x)
347        }
348        Err(e) => {
349            writer.inner.reset(e.code()).ok();
350            writer.transfer_aborted().await;
351            Err(e)
352        }
353    }
354}
355async fn handle_read_result<R: RecvStream, T, E: HasErrorCode>(
356    reader: &mut ProgressReader<R>,
357    r: Result<T, E>,
358) -> Result<T, E> {
359    match r {
360        Ok(x) => {
361            reader.transfer_completed().await;
362            Ok(x)
363        }
364        Err(e) => {
365            reader.inner.stop(e.code()).ok();
366            reader.transfer_aborted().await;
367            Err(e)
368        }
369    }
370}

372pub async fn handle_stream<R: RecvStream, W: SendStream>(
373    pair: StreamPair<R, W>,
374    store: Store,
375) -> n0_error::Result<()> {
376    let result = dispatch(pair, store).await;
377    if let Err(error) = &result {
378        // The peer only sees a reset with an error code, so keep the cause.
379        debug!("stream failed: {error:#}");
380    }
381    result
382}
383
384async fn dispatch<R: RecvStream, W: SendStream>(
385    mut pair: StreamPair<R, W>,
386    store: Store,
387) -> n0_error::Result<()> {
388    let request = pair.read_request().await?;
389    match request {
390        Request::Get(request) => handle_get(pair, store, request).await?,
391        Request::GetMany(request) => handle_get_many(pair, store, request).await?,
392        Request::Observe(request) => handle_observe(pair, store, request).await?,
393        Request::Push(request) => handle_push(pair, store, request).await?,
394        _ => {}
395    }
396    Ok(())
397}

399#[stack_error(derive, add_meta, from_sources)]
400pub enum HandleGetError {
401    #[error(transparent)]
402    ExportBao {
403        #[error(std_err)]
404        source: ExportBaoError,
405    },
406    #[error("Invalid hash sequence")]
407    InvalidHashSeq {},
408    #[error("Invalid offset")]
409    InvalidOffset {},
410}
411
412impl HasErrorCode for HandleGetError {
413    fn code(&self) -> VarInt {
414        match self {
415            HandleGetError::ExportBao {
416                source: ExportBaoError::ClientError { source, .. },
417                ..
418            } => source.code(),
419            HandleGetError::InvalidHashSeq { .. } => ERR_INTERNAL,
420            HandleGetError::InvalidOffset { .. } => ERR_INTERNAL,
421            _ => ERR_INTERNAL,
422        }
423    }
424}
425
426impl HandleGetError {
427    /// Whether the request failed because of a problem on our side.
428    fn is_local_failure(&self) -> bool {
429        match self {
430            Self::ExportBao { source, .. } => is_local_export_failure(source),
431            Self::InvalidHashSeq { .. } | Self::InvalidOffset { .. } => false,
432        }
433    }
434}
435
436impl From<LazyHashSeqError> for HandleGetError {
437    fn from(e: LazyHashSeqError) -> Self {
438        match e {
439            LazyHashSeqError::InvalidHashSeq { .. } => e!(HandleGetError::InvalidHashSeq),
440            LazyHashSeqError::Request { source, .. } => {
441                let source = match source {
442                    RequestError::Rpc { source, .. } => source.into(),
443                    // a store error, so keep it apart from network errors
444                    RequestError::Inner {
445                        source: crate::api::Error::Io(e),
446                        ..
447                    } => e!(ExportBaoError::ExportBaoInner, EncodeError::Io(e)),
448                };
449                e!(HandleGetError::ExportBao, source)
450            }
451        }
452    }
453}

455/// Whether an export failed because of a problem on our side.
456///
457/// An io error means writing to the stream failed, so the peer went away. A
458/// client error means the event handler aborted the request. And `NotFound`
459/// means we don't have the requested blob.
460fn is_local_export_failure(e: &ExportBaoError) -> bool {
461    match e {
462        ExportBaoError::ExportBaoIo { .. } | ExportBaoError::ClientError { .. } => false,
463        ExportBaoError::ExportBaoInner {
464            source: EncodeError::Io(e),
465            ..
466        } => e.kind() != io::ErrorKind::NotFound,
467        _ => true,
468    }
469}

471/// Handle a single get request.
472///
473/// Requires a database, the request, and a writer.
474async fn handle_get_impl<W: SendStream>(
475    store: Store,
476    request: GetRequest,
477    writer: &mut ProgressWriter<W>,
478) -> Result<(), HandleGetError> {
479    let hash = request.hash;
480    debug!(%hash, "get received request");
481    let mut hash_seq = LazyHashSeq::new(store.blobs().clone(), hash);
482    for (offset, ranges) in request.ranges.iter_non_empty_infinite() {
483        if offset == 0 {
484            send_blob(&store, offset, hash, ranges.clone(), writer).await?;
485        } else {
486            let Some(hash) = hash_seq.get(offset - 1).await? else {
487                break;
488            };
489            send_blob(&store, offset, hash, ranges.clone(), writer).await?;
490        }
491    }
492    writer
493        .inner
494        .sync()
495        .await
496        .map_err(|e| e!(HandleGetError::ExportBao, e.into()))?;
497
498    Ok(())
499}

501pub async fn handle_get<R: RecvStream, W: SendStream>(
502    mut pair: StreamPair<R, W>,
503    store: Store,
504    request: GetRequest,
505) -> n0_error::Result<()> {
506    let res = pair.get_request(|| request.clone()).await;
507    let tracker = handle_read_request_result(&mut pair, res).await?;
508    let mut writer = pair.into_writer(tracker).await?;
509    let res = handle_get_impl(store, request, &mut writer).await;
510    if let Err(e) = &res {
511        if e.is_local_failure() {
512            warn!("get request failed: {e:#}");
513        }
514    }
515    handle_write_result(&mut writer, res).await?;
516    Ok(())
517}

546/// Handle a single get request.
547///
548/// Requires a database, the request, and a writer.
549async fn handle_get_many_impl<W: SendStream>(
550    store: Store,
551    request: GetManyRequest,
552    writer: &mut ProgressWriter<W>,
553) -> Result<(), HandleGetManyError> {
554    debug!("get_many received request");
555    let request_ranges = request.ranges.iter_infinite();
556    for (child, (hash, ranges)) in request.hashes.iter().zip(request_ranges).enumerate() {
557        if !ranges.is_empty() {
558            send_blob(&store, child as u64, *hash, ranges.clone(), writer).await?;
559        }
560    }
561    Ok(())
562}

615/// Handle a single push request.
616///
617/// Requires a database, the request, and a reader.
618async fn handle_push_impl<R: RecvStream>(
619    store: Store,
620    request: PushRequest,
621    reader: &mut ProgressReader<R>,
622) -> Result<(), HandlePushError> {
623    let hash = request.hash;
624    debug!(%hash, "push received request");
625    let root_ranges = request
626        .ranges
627        .iter_infinite()
628        .next()
629        .expect("infinite iterator");
630    if !root_ranges.is_empty() {
631        // todo: send progress from import_bao_noq or rename to import_bao_noq_with_progress
632        store
633            .import_bao_reader(hash, root_ranges.clone(), &mut reader.inner)
634            .await?;
635    }
636    if request.ranges.is_blob() {
637        debug!("push request complete");
638        return Ok(());
639    }
640    let mut hash_seq = LazyHashSeq::new(store.blobs().clone(), hash);
641    for (offset, child_ranges) in request.ranges.iter_non_empty_infinite() {
642        if offset == 0 {
643            continue;
644        }
645        let Some(child_hash) = hash_seq.get(offset - 1).await? else {
646            break;
647        };
648        store
649            .import_bao_reader(child_hash, child_ranges.clone(), &mut reader.inner)
650            .await?;
651    }
652    Ok(())
653}
654
655pub async fn handle_push<R: RecvStream, W: SendStream>(
656    mut pair: StreamPair<R, W>,
657    store: Store,
658    request: PushRequest,
659) -> n0_error::Result<()> {
660    let res = pair.push_request(|| request.clone()).await;
661    let tracker = handle_read_request_result(&mut pair, res).await?;
662    let mut reader = pair.into_reader(tracker).await?;
663    let res = handle_push_impl(store, request, &mut reader).await;
664    handle_read_result(&mut reader, res).await?;
665    Ok(())
666}

707/// Handle a single push request.
708///
709/// Requires a database, the request, and a reader.
710async fn handle_observe_impl<W: SendStream>(
711    store: Store,
712    request: ObserveRequest,
713    writer: &mut ProgressWriter<W>,
714) -> std::result::Result<(), HandleObserveError> {
715    let mut stream = store
716        .observe(request.hash)
717        .stream()
718        .await
719        .map_err(|_| e!(HandleObserveError::ObserveStreamClosed))?;
720    let mut old = stream
721        .next()
722        .await
723        .ok_or_else(|| e!(HandleObserveError::ObserveStreamClosed))?;
724    // send the initial bitfield
725    send_observe_item(writer, &old).await?;
726    // send updates until the remote loses interest
727    loop {
728        select! {
729            new = stream.next() => {
730                let new = new.ok_or_else(|| e!(HandleObserveError::ObserveStreamClosed))?;
731                let diff = old.diff(&new);
732                if diff.is_empty() {
733                    continue;
734                }
735                send_observe_item(writer, &diff).await?;
736                old = new;
737            }
738            _ = writer.inner.stopped() => {
739                debug!("observer closed");
740                break;
741            }
742        }
743    }
744    Ok(())
745}
746
747async fn send_observe_item<W: SendStream>(
748    writer: &mut ProgressWriter<W>,
749    item: &Bitfield,
750) -> io::Result<()> {
751    let item = ObserveItem::from(item);
752    let len = writer.inner.write_length_prefixed(item).await?;
753    writer.context.log_other_write(len);
754    Ok(())
755}
756
757pub async fn handle_observe<R: RecvStream, W: SendStream>(
758    mut pair: StreamPair<R, W>,
759    store: Store,
760    request: ObserveRequest,
761) -> n0_error::Result<()> {
762    let res = pair.observe_request(|| request.clone()).await;
763    let tracker = handle_read_request_result(&mut pair, res).await?;
764    let mut writer = pair.into_writer(tracker).await?;
765    let res = handle_observe_impl(store, request, &mut writer).await;
766    if let Err(e) = &res {
767        if e.is_local_failure() {
768            warn!("observe request failed: {e:#}");
769        }
770    }
771    handle_write_result(&mut writer, res).await?;
772    Ok(())
773}
