14    Data,

16    Bitfield,

19const ALL_PARTS: [BaoFilePart; 4] = [
20    BaoFilePart::Outboard,
21    BaoFilePart::Data,
22    BaoFilePart::Sizes,
23    BaoFilePart::Bitfield,
24];

26/// The owned files of an entry that the given state does not reference.
27///
28/// This may include files that do not exist. Deleting those is a no-op.
29pub(super) fn unreferenced_parts<I>(state: Option<&EntryState<I>>) -> Vec<BaoFilePart> {
30    match state {
31        None => ALL_PARTS.to_vec(),
32        Some(EntryState::Partial { .. }) => Vec::new(),
33        Some(EntryState::Complete {
34            data_location,
35            outboard_location,
36        }) => {
37            let mut parts = vec![BaoFilePart::Sizes, BaoFilePart::Bitfield];
38            if !matches!(data_location, DataLocation::Owned(_)) {
39                parts.push(BaoFilePart::Data);
40            }
41            if !matches!(outboard_location, OutboardLocation::Owned) {
42                parts.push(BaoFilePart::Outboard);
43            }
44            parts
45        }
46    }
47}

49/// Which hashes have a live entity, as reported by the entities themselves.
50///
51/// An entity is marked live by the load that precedes its first file access,
52/// and reports itself not live at the end of its shutdown. An entity that only
53/// reads a complete entry never touches its files' contents in a way that
54/// matters here, so it is never marked live. Both go through the metadata
55/// FIFO, so:
56///
57/// - an entity only touches files of its hash after the metadata actor has
58///   answered its load, so after it was marked live, and
59/// - when its inactive is processed, all its metadata writes have been
60///   processed before, and it touches no more files.
61///
62/// So at the end of a batch, the entry state of a hash that is not live
63/// reflects all files of that hash.
64#[derive(Debug, Default)]
65pub(super) struct LiveEntities(HashSet<Hash>);
66
67impl LiveEntities {
68    pub fn load(&mut self, hash: Hash) {
69        self.0.insert(hash);
70    }
71
72    pub fn inactive(&mut self, hash: Hash) {
73        self.0.remove(&hash);
74    }
75
76    pub fn is_live(&self, hash: &Hash) -> bool {
77        self.0.contains(hash)
78    }
79}

81/// Delete the given files of an entry, ignoring files that do not exist.
82pub(super) fn delete_files(options: &PathOptions, hash: &Hash, parts: &[BaoFilePart]) {
83    for part in parts {
84        let path = match part {
85            BaoFilePart::Data => options.data_path(hash),
86            BaoFilePart::Outboard => options.outboard_path(hash),
87            BaoFilePart::Sizes => options.sizes_path(hash),
88            BaoFilePart::Bitfield => options.bitfield_path(hash),
89        };
90        if let Err(cause) = std::fs::remove_file(&path) {
91            // Ignore NotFound errors, if the file is already gone that's fine.
92            if cause.kind() != std::io::ErrorKind::NotFound {
93                warn!("failed to delete {part:?} {}: {cause}", path.display());
94            }
95        }
96    }
97}
