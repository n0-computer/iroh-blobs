1//! Where the fs store keeps the bytes of its entries.
2//!
3//! The store keeps all the logic: which parts an entry has, bao validation,
4//! inlining, crash consistency. A [`Storage`] only maps a hash's parts to
5//! locations, and opens and deletes the resources there.
6//!
7//! A location describes where bytes live, a `PathBuf` for files. A resource is
8//! an open handle to them, a `File` today, a memory map for complete parts
9//! later. The location type is fixed per store type, so a store never has to
10//! tell kinds of locations apart.
11//!
12//! Draft: the store still uses [`PathOptions`] as its only storage. Staging
13//! imports, syncing, external data, bitfields and recovery still use paths
14//! directly.
15

31/// How to open a resource.
32#[derive(Debug, Clone, Copy, PartialEq, Eq)]
33pub(crate) enum Mode {
34    /// Read a part of a complete entry, which is never written again.
35    Read,
36    /// Read and write a part of a partial entry, creating it if missing.
37    ReadWrite,
38}
39
40/// Maps the parts of an entry to locations, and opens and deletes the resources there.
41pub(crate) trait Storage: fmt::Debug + Send + Sync + 'static {
42    /// Describes where bytes live.
43    type Location: Clone + fmt::Debug + Send + Sync + 'static;
44    /// An open handle to the bytes at a location.
45    type Resource: ReadAt + ReadBytesAt + WriteAt + Size + fmt::Debug + Send + Sync + 'static;
46
47    /// The location of a part of the entry for `hash`.
48    fn location(&self, hash: &Hash, part: BaoFilePart) -> Self::Location;
49
50    /// Opens the resource at `location`.
51    fn open(&self, location: &Self::Location, mode: Mode) -> io::Result<Self::Resource>;
52
53    /// Deletes the resource at `location`; one that does not exist is not an error.
54    fn delete(&self, location: &Self::Location) -> io::Result<()>;
55
56    /// Opens a part of the entry for `hash`.
57    fn open_part(&self, hash: &Hash, part: BaoFilePart, mode: Mode) -> io::Result<Self::Resource> {
58        self.open(&self.location(hash, part), mode)
59    }
60
61    /// Deletes a part of the entry for `hash`; one that does not exist is not an error.
62    fn delete_part(&self, hash: &Hash, part: BaoFilePart) -> io::Result<()> {
63        self.delete(&self.location(hash, part))
64    }
65}
66
67/// Plain files in the store's data directory.
68impl Storage for PathOptions {
69    type Location = PathBuf;
70    type Resource = File;
71
72    fn location(&self, hash: &Hash, part: BaoFilePart) -> PathBuf {
73        match part {
74            BaoFilePart::Data => self.data_path(hash),
75            BaoFilePart::Outboard => self.outboard_path(hash),
76            BaoFilePart::Sizes => self.sizes_path(hash),
77            BaoFilePart::Bitfield => self.bitfield_path(hash),
78        }
79    }
80
81    fn open(&self, location: &PathBuf, mode: Mode) -> io::Result<File> {
82        match mode {
83            Mode::Read => File::open(location),
84            // without truncating, the part may hold data already
85            Mode::ReadWrite => OpenOptions::new()
86                .read(true)
87                .write(true)
88                .create(true)
89                .truncate(false)
90                .open(location),
91        }
92    }
93
94    fn delete(&self, location: &PathBuf) -> io::Result<()> {
95        match fs::remove_file(location) {
96            Err(cause) if cause.kind() == io::ErrorKind::NotFound => Ok(()),
97            res => res,
98        }
99    }
100}
