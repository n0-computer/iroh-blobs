1//! The metadata protocol: what a metadata store answers, independent of how it stores it.
2//!
3//! A metadata store is one actor that receives [`Request`]s on one ordered
4//! channel. Handlers send the per-entry requests for their own hash, and the
5//! main actor sends the store-wide ones. The actor sends unreferenced parts
6//! back to the store after each commit, and fails the store by ending with an
7//! error.
8//!
9//! How the actor batches requests into transactions is up to it. The fs store's
10//! redb actor converts each request into its own command grouping as it
11//! receives it, and groups them by that.
12//!
13//! # Contract
14//!
15//! - **One FIFO.** Each request sees the effect of every earlier one, also
16//!   before it is committed, and no request overtakes another.
17//! - **Applied or committed.** `Set` and `UpdateAwait` are answered once
18//!   applied, which means visible to later requests. `Delete` and `SyncDb` are
19//!   answered once committed, also a `Delete` of an absent entry. `Update` is
20//!   not answered.
21//! - **Failure.** If a batch fails, its outstanding replies are dropped and
22//!   the actor ends with an error, which aborts the store. Nothing keeps
23//!   running on an applied but lost state.
24//! - **Listings.** `ListTags`, `ListBlobs` and `GcSnapshot` each read one
25//!   consistent snapshot, taken at their place in the FIFO, and reply with a
26//!   stream that owns it, so the actor never waits for the reader.
27//! - **Unreferenced parts** are sent only after the commit that stopped
28//!   referencing them, on an unbounded channel, so the actor never waits on the
29//!   store.
30//! - **Shutdown** commits everything, records the clean shutdown so the next
31//!   open needs no recovery, and replies. Then the actor ends.

67/// Why a metadata store ended, and what it keeps alive until the store has stopped its workers.
68///
69/// The redb store keeps its database here, so the database stays locked
70/// until no worker can touch files anymore.
71pub(crate) struct Failure {
72    pub error: n0_error::AnyError,
73    _guard: Box<dyn Any + Send>,
74}
75
76impl Failure {
77    pub fn new(error: n0_error::AnyError, guard: impl Any + Send) -> Self {
78        Self {
79            error,
80            _guard: Box::new(guard),
81        }
82    }
83}
84
85impl fmt::Debug for Failure {
86    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
87        f.debug_struct("Failure")
88            .field("error", &self.error)
89            .finish_non_exhaustive()
90    }
91}

93/// A request to the metadata store.
94#[derive(Debug)]
95#[enum_conversions()]
96pub(crate) enum Request {
97    // Per entry, sent by the hash's handler.
98    /// Get the entry state, with inline data and outboard.
99    Get(Get),
100    /// Replace the entry state. Answered once applied.
101    Set(Set),
102    /// Merge into the entry state. Not answered.
103    Update(Update),
104    /// Merge into the entry state. Answered once applied.
105    UpdateAwait(UpdateAwait),
106    /// Make sure the entry is absent. Answered once committed.
107    Delete(Delete),
108
109    // Store-wide, sent by the main actor.
110    /// Set a tag, or remove it.
111    SetTag(SetTagMsg),
112    /// Create a new unique tag.
113    CreateTag(CreateTagMsg),
114    /// Remove a range of tags.
115    DeleteTags(DeleteTagsMsg),
116    /// Rename a tag.
117    RenameTag(RenameTagMsg),
118    /// List tags from a snapshot.
119    ListTags(ListTags),
120    /// The status of a blob.
121    BlobStatus(BlobStatusMsg),
122    /// List the hashes of all blobs from a snapshot.
123    ListBlobs(ListBlobs),
124    /// The gc candidates and roots from one snapshot.
125    GcSnapshot(GcSnapshot),
126    /// Answered once everything before it is committed.
127    SyncDb(SyncDbMsg),
128    /// Commit, record the clean shutdown, reply, and end.
129    Shutdown(ShutdownMsg),
130}
