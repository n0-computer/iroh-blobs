15/// Get the entry state for a hash.
16///
17/// This will read from the blobs table and enrich the result with the content
18/// of the inline data and inline outboard tables if necessary.
19pub struct Get {
20    pub hash: Hash,
21    pub tx: oneshot::Sender<GetResult>,
22    pub span: Span,
23}
24
25impl fmt::Debug for Get {
26    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
27        f.debug_struct("Get")
28            .field("hash", &DD(self.hash.to_hex()))
29            .finish_non_exhaustive()
30    }
31}

33/// Delete the entry of one hash, sent by the hash's handler.
34///
35/// The handler sends this from its `Deleting` state, so none of its writes can
36/// follow. The reply comes once the delete is committed, even if no entry
37/// existed, and it also detects commands dropped during shutdown or a failed
38/// batch, since then it is dropped.
39/// Whether the entry may be deleted is decided before, by the main actor's
40/// temp tag check. A metadata error fails the store instead of undoing a delete.
41#[derive(Debug)]
42pub struct Delete {
43    pub hash: Hash,
44    pub tx: oneshot::Sender<io::Result<()>>,
45    pub span: Span,
46}
47
48#[derive(Debug)]
49pub struct GetResult {
50    pub state: io::Result<Option<EntryState<Bytes>>>,
51}

53/// A stream of rows from a read snapshot, which it owns.
54///
55/// The metadata actor hands it out and moves on, so it never waits for the reader.
56pub type Listing<T> = Pin<Box<dyn Stream<Item = api::Result<T>> + Send + 'static>>;
57
58/// The gc candidates (every blob) and the named tag roots, from one snapshot.
59pub type GcSnapshotReply = (Listing<Hash>, Listing<HashAndFormat>);

100/// Merge into the entry state of a hash, without a reply.
101///
102/// Sent by the hash's handler for its writes. If applying it fails, the
103/// metadata actor fails, and with it the store, so the handler never runs on
104/// a state the metadata does not have.
105pub struct Update {
106    pub hash: Hash,
107    pub state: EntryState<Bytes>,
108    pub span: Span,
109}
110
111impl fmt::Debug for Update {
112    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
113        f.debug_struct("Update")
114            .field("hash", &self.hash)
115            .field("state", &DD(self.state.fmt_short()))
116            .finish_non_exhaustive()
117    }
118}

120/// Merge into the entry state of a hash, replied once applied.
