13/// Two states of one entry that cannot both be true, such as two different validated sizes.
14#[derive(Debug)]
15pub struct InconsistentEntry(pub String);
16
17impl fmt::Display for InconsistentEntry {
18    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
19        write!(f, "inconsistent entry: {}", self.0)
20    }
21}
22
23impl std::error::Error for InconsistentEntry {}

25/// Where a backend keeps data outside the store, in a format only it interprets.
26///
27/// The host never looks inside. The fs backend keeps a UTF-8 path. It is
28/// encoded like a string, the format of external paths in stored entries.
29#[derive(Clone, PartialEq, Eq, PartialOrd, Ord, Hash, Serialize, Deserialize)]
30#[serde(transparent)]
31pub struct Location(Bytes);
32
33impl Location {
34    /// A location in a backend's own format.
35    pub fn new(location: impl Into<Bytes>) -> Self {
36        Self(location.into())
37    }
38
39    /// The location in the backend's own format.
40    pub fn as_bytes(&self) -> &[u8] {
41        &self.0
42    }
43
44    /// The location of a file, for backends that keep paths.
45    ///
46    /// Fails for a path that is not UTF-8, the only kind every platform can read back.
47    pub fn from_path(path: &Path) -> io::Result<Self> {
48        let path = path.to_str().ok_or_else(|| {
49            io::Error::new(
50                io::ErrorKind::InvalidInput,
51                format!("path is not UTF-8: {}", path.display()),
52            )
53        })?;
54        Ok(Self(Bytes::copy_from_slice(path.as_bytes())))
55    }
56
57    /// The location as a file path, for backends that keep paths.
58    pub fn to_path(&self) -> io::Result<&Path> {
59        let path = std::str::from_utf8(&self.0)
60            .map_err(|_| io::Error::new(io::ErrorKind::InvalidData, "location is not UTF-8"))?;
61        Ok(Path::new(path))
62    }
63}
64
65impl Debug for Location {
66    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
67        match std::str::from_utf8(&self.0) {
68            Ok(s) => Debug::fmt(s, f),
69            Err(_) => write!(f, "{:?}", self.0),
70        }
71    }
72}

74/// Location of the data.
75///
76/// Data can be inlined in the database, a file conceptually owned by the store,
77/// or a number of external files conceptually owned by the user.
78///
79/// Only complete data can be inlined.
80#[derive(Debug, Clone, Serialize, Deserialize, PartialEq, Eq)]
81pub enum DataLocation<I = (), E = ()> {
82    /// Data is in the inline_data table.
83    Inline(I),
84    /// Data is in the canonical location in the data directory.
85    Owned(E),
86    /// Data is in several external locations. This should be a non-empty list.
87    External(Vec<Location>, E),
88}
89
90impl<I: AsRef<[u8]>, E: Debug> DataLocation<I, E> {
91    fn fmt_short(&self) -> String {
92        match self {
93            DataLocation::Inline(d) => {
94                format!("Inline({}, addr={})", d.as_ref().len(), d.addr_short())
95            }
96            DataLocation::Owned(e) => format!("Owned({e:?})"),
97            DataLocation::External(paths, e) => format!("External({paths:?}, {e:?})"),
98        }
99    }
100}
101
102impl<X> DataLocation<X, u64> {
103    #[allow(clippy::result_large_err)]
104    fn union(self, that: DataLocation<X, u64>) -> Result<Self, InconsistentEntry> {
105        Ok(match (self, that) {
106            (
107                DataLocation::External(mut paths, a_size),
108                DataLocation::External(b_paths, b_size),
109            ) => {
110                if a_size != b_size {
111                    return Err(InconsistentEntry(format!(
112                        "complete size mismatch {a_size} {b_size}"
113                    )));
114                }
115                paths.extend(b_paths);
116                paths.sort();
117                paths.dedup();
118                DataLocation::External(paths, a_size)
119            }
120            (_, b @ DataLocation::Owned(_)) => {
121                // owned needs to win, since it has an associated file. Choosing
122                // external would orphan the file.
123                b
124            }
125            (a @ DataLocation::Owned(_), _) => {
126                // owned needs to win, since it has an associated file. Choosing
127                // external would orphan the file.
128                a
129            }
130            (_, b @ DataLocation::Inline(_)) => {
131                // inline needs to win, since it has associated data. Choosing
132                // external would orphan the file.
133                b
134            }
135            (a @ DataLocation::Inline(_), _) => {
136                // inline needs to win, since it has associated data. Choosing
137                // external would orphan the file.
138                a
139            }
140        })
141    }
142}

144impl<I, E> DataLocation<I, E> {
145    pub fn split_inline_data(self) -> (DataLocation<(), E>, Option<I>) {
146        match self {
147            DataLocation::Inline(x) => (DataLocation::Inline(()), Some(x)),
148            DataLocation::Owned(x) => (DataLocation::Owned(x), None),
149            DataLocation::External(paths, x) => (DataLocation::External(paths, x), None),
150        }
151    }
152}

154/// Location of the outboard.
155///
156/// Outboard can be inlined in the database or a file conceptually owned by the store.
157/// Outboards are implementation specific to the store and as such are always owned.
158///
159/// Only complete outboards can be inlined.
160#[derive(Debug, Clone, Serialize, Deserialize, PartialEq, Eq)]
161pub enum OutboardLocation<I = ()> {
162    /// Outboard is in the inline_outboard table.
163    Inline(I),
164    /// Outboard is in the canonical location in the data directory.
165    Owned,
166    /// Outboard is not needed
167    NotNeeded,
168}
169
170impl<I: AsRef<[u8]>> OutboardLocation<I> {
171    fn fmt_short(&self) -> String {
172        match self {
173            OutboardLocation::Inline(d) => format!("Inline({})", d.as_ref().len()),
174            OutboardLocation::Owned => "Owned".to_string(),
175            OutboardLocation::NotNeeded => "NotNeeded".to_string(),
176        }
177    }
178}
179
180impl<I> OutboardLocation<I> {
181    pub fn inline(data: I) -> Self
182    where
183        I: AsRef<[u8]>,
184    {
185        if data.as_ref().is_empty() {
186            OutboardLocation::NotNeeded
187        } else {
188            OutboardLocation::Inline(data)
189        }
190    }
191
192    pub fn split_inline_data(self) -> (OutboardLocation<()>, Option<I>) {
193        match self {
194            Self::Inline(x) => (OutboardLocation::Inline(()), Some(x)),
195            Self::Owned => (OutboardLocation::Owned, None),
196            Self::NotNeeded => (OutboardLocation::NotNeeded, None),
197        }
198    }
199}

201/// The information about an entry that we keep in the entry table for quick access.
202///
203/// The exact info to store here is TBD, so usually you should use the accessor methods.
204#[derive(Debug, Clone, Serialize, Deserialize, PartialEq, Eq)]
205pub enum EntryState<I = ()> {
206    /// For a complete entry we always know the size. It does not make much sense
207    /// to write to a complete entry, so they are much easier to share.
208    Complete {
209        /// Location of the data.
210        data_location: DataLocation<I, u64>,
211        /// Location of the outboard.
212        outboard_location: OutboardLocation<I>,
213    },
214    /// Partial entries are entries for which we know the hash, but don't have
215    /// all the data. They are created when syncing from somewhere else by hash.
216    ///
217    /// As such they are always owned. There is also no inline storage for them.
218    /// Non short lived partial entries always live in the file system, and for
219    /// short lived ones we never create a database entry in the first place.
220    Partial {
221        /// Once we get the last chunk of a partial entry, we have validated
222        /// the size of the entry despite it still being incomplete.
223        ///
224        /// E.g. a giant file where we just requested the last chunk.
225        size: Option<u64>,
226    },
227}
228
229impl<I> EntryState<I> {
230    pub fn is_complete(&self) -> bool {
231        matches!(self, Self::Complete { .. })
232    }
233
234    pub fn is_partial(&self) -> bool {
235        matches!(self, Self::Partial { .. })
236    }
237}
238
239impl Default for EntryState {
240    fn default() -> Self {
241        Self::Partial { size: None }
242    }
243}
244
245impl<I: AsRef<[u8]>> EntryState<I> {
246    pub fn fmt_short(&self) -> String {
247        match self {
248            Self::Complete {
249                data_location,
250                outboard_location,
251            } => format!(
252                "Complete {{ data: {}, outboard: {} }}",
253                data_location.fmt_short(),
254                outboard_location.fmt_short()
255            ),
256            Self::Partial { size } => format!("Partial {{ size: {size:?} }}"),
257        }
258    }
259}
260
261impl EntryState {
262    pub fn union(old: Self, new: Self) -> Result<Self, InconsistentEntry> {
263        match (old, new) {
264            (
265                Self::Complete {
266                    data_location,
267                    outboard_location,
268                },
269                Self::Complete {
270                    data_location: b_data_location,
271                    ..
272                },
273            ) => Ok(Self::Complete {
274                // combine external paths if needed
275                data_location: data_location.union(b_data_location)?,
276                outboard_location,
277            }),
278            (a @ Self::Complete { .. }, Self::Partial { .. }) =>
279            // complete wins over partial
280            {
281                Ok(a)
282            }
283            (Self::Partial { .. }, b @ Self::Complete { .. }) =>
284            // complete wins over partial
285            {
286                Ok(b)
287            }
288            (Self::Partial { size: a_size }, Self::Partial { size: b_size }) =>
289            // keep known size from either entry
290            {
291                let size = match (a_size, b_size) {
292                    (Some(a_size), Some(b_size)) => {
293                        // validated sizes are different. this means that at
294                        // least one validation was wrong, which would be a bug
295                        // in bao-tree.
296                        if a_size != b_size {
297                            return Err(InconsistentEntry(format!(
298                                "validated size mismatch {a_size} {b_size}"
299                            )));
300                        }
301                        Some(a_size)
302                    }
303                    (Some(a_size), None) => Some(a_size),
304                    (None, Some(b_size)) => Some(b_size),
305                    (None, None) => None,
306                };
307                Ok(Self::Partial { size })
308            }
309        }
310    }
311}
