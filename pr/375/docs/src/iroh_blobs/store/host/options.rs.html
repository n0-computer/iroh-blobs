7/// The part of the fs store's `Options` the store itself uses.
8///
9/// The rest goes to the parts the store is built from: the fs handler gets its
10/// paths and sync mode as `FsConfig`, and the redb metadata store gets the
11/// options it needs when it is created.
12#[derive(Debug, Clone)]
13pub(crate) struct StoreOptions {
14    /// Inlining, which every handler implementation shares.
15    pub inline: InlineOptions,
16    /// Where imports stage their files.
17    ///
18    /// Imports stage and hash their data before the hash is known, so outside
19    /// any handler, and the fs handler then renames the files into place. So
20    /// this belongs to the fs handler really, and must be on the same device as
21    /// its data directory. It moves there once import staging is part of the
22    /// handler implementation.
23    pub temp_path: PathBuf,
24    /// Gc configuration.
25    pub gc: Option<GcConfig>,
26}
27
28impl StoreOptions {
29    /// Check if the data will be inlined, based on the size of the data.
30    pub fn is_inlined_data(&self, data_size: u64) -> bool {
31        data_size <= self.inline.max_data_inlined
32    }
33
34    /// Check if the outboard will be inlined, based on the size of the *outboard*.
35    pub fn is_inlined_outboard(&self, outboard_size: u64) -> bool {
36        outboard_size <= self.inline.max_outboard_inlined
37    }
38
39    /// Check if both the data and outboard will be inlined, based on the size of the data.
40    pub fn is_inlined_all(&self, data_size: u64) -> bool {
41        let outboard_size = raw_outboard_size(data_size);
42        self.is_inlined_data(data_size) && self.is_inlined_outboard(outboard_size)
43    }
44
45    /// A new, unique file name in the temp directory.
46    pub fn temp_file_name(&self) -> PathBuf {
47        self.temp_path.join(temp_name())
48    }
49}
50
51/// Create a 16 byte unique ID.
52fn new_uuid() -> [u8; 16] {
53    use rand::Rng;
54    let mut rng = rand::rng();
55    let mut bytes = [0u8; 16];
56    rng.fill_bytes(&mut bytes);
57    bytes
58}
59
60/// Create temp file name based on a 16 byte UUID.
61pub(crate) fn temp_name() -> String {
62    format!("{}.temp", hex::encode(new_uuid()))
63}

65/// Options for inlining small complete data or outboards.
66#[derive(Debug, Clone)]
67pub struct InlineOptions {
68    /// Maximum data size to inline.
69    pub max_data_inlined: u64,
70    /// Maximum outboard size to inline.
71    pub max_outboard_inlined: u64,
72}
73
74impl InlineOptions {
75    /// Do not inline anything, ever.
76    pub const NO_INLINE: Self = Self {
77        max_data_inlined: 0,
78        max_outboard_inlined: 0,
79    };
80    /// Always inline everything
81    pub const ALWAYS_INLINE: Self = Self {
82        max_data_inlined: u64::MAX,
83        max_outboard_inlined: u64::MAX,
84    };
85}
86
87impl Default for InlineOptions {
88    fn default() -> Self {
89        Self {
90            max_data_inlined: 1024 * 16,
91            max_outboard_inlined: 1024 * 16,
92        }
93    }
