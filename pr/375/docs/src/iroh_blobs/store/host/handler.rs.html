1//! What a store is built from: a [`Backend`], and the handlers it runs.
2//!
3//! A backend names the handler type and starts the metadata store, the parts
4//! the host does not run itself.
5//!
6//! A handler is the state of one hash. The host creates it with
7//! [`HashHandler::new`] when a request for the hash arrives, and recycles it with
8//! [`HashHandler::reset`] for another hash once it is idle, keeping its allocations.
9//! Neither may do IO: loading is lazy, in the first request that needs it.
10//!
11//! Each request runs as a task on the handler's actor. Tasks of one handler
12//! interleave at `.await` points but never run in parallel. Every method gets
13//! the store-wide [`HandlerGlobal`]: the store's context, and the backend value,
14//! which is the handlers' configuration.
15//!
16//! # Contract
17//!
18//! - The handler is the only writer of its entry's metadata and the only one
19//!   that touches its files.
20//! - It decides about files only from loaded state: a delete or an
21//!   unreferenced-parts notice loads first.
22//! - A delete starts a new generation, which ends observers, and is answered
23//!   once committed. Writes wait until it is done.
24//! - While frozen, writes and deletes wait; reads go on.
25//! - [`HashHandler::persist`] runs when the handler unloads, idle or at shutdown.

41/// The parts a store is built from, beyond its own options.
42pub(crate) trait Backend: fmt::Debug + Send + Sync + 'static {
43    /// The state of one hash, which handles its requests.
44    ///
45    /// The host creates it with [`HashHandler::new`], so this type is the
46    /// handler factory.
47    type Handler: HashHandler<Backend = Self>;
48
49    /// Starts the metadata store with its channels, on the store's runtime.
50    ///
51    /// Returns it running. It ends with an error to fail the store.
52    fn start_meta(&self, channels: MetaChannels) -> n0_error::Result<impl MetaTask>;
53}

55/// What every handler of a store shares.
56#[derive(Debug)]
57pub(crate) struct HandlerGlobal<B> {
58    /// The store: options, and the client of the metadata store.
59    pub store: Arc<TaskContext>,
60    /// The backend, which is the handlers' configuration.
61    pub backend: B,
62}

64/// The state of one hash, which handles its requests.
65pub(crate) trait HashHandler: fmt::Debug + Clone + Send + Sync + 'static {
66    /// The backend this handler belongs to.
67    type Backend: Backend<Handler = Self>;
68
69    /// A new handler for `hash`. Must be cheap and do no IO.
70    fn new(hash: &Hash) -> Self;
71
72    /// Rebinds a recycled handler to `hash`.
73    ///
74    /// Drops what it held for its previous hash, but keeps its allocations.
75    fn reset(&mut self, hash: &Hash);
76
77    /// Import from a stream of n0 bao encoded data.
78    fn import_bao(
79        &self,
80        global: &HandlerGlobal<Self::Backend>,
81        cmd: ImportBaoMsg,
82    ) -> impl Future<Output = ()> + Send;
83
84    /// Finish an import from a local file or memory.
85    fn finish_import(
86        &self,
87        global: &HandlerGlobal<Self::Backend>,
88        cmd: ImportEntryMsg,
89        tt: TempTag,
90    ) -> impl Future<Output = ()> + Send;
91
92    /// Observe the bitfield of the entry, until it is deleted.
93    fn observe(
94        &self,
95        global: &HandlerGlobal<Self::Backend>,
96        cmd: ObserveMsg,
97    ) -> impl Future<Output = ()> + Send;
98
99    /// Export byte ranges of the entry.
100    fn export_ranges(
101        &self,
102        global: &HandlerGlobal<Self::Backend>,
103        cmd: ExportRangesMsg,
104    ) -> impl Future<Output = ()> + Send;
105
106    /// Export chunk ranges of the entry as a n0 bao encoded stream.
107    fn export_bao(
108        &self,
109        global: &HandlerGlobal<Self::Backend>,
110        cmd: ExportBaoMsg,
111    ) -> impl Future<Output = ()> + Send;
112
113    /// Export the entry to a local file.
114    fn export_path(
115        &self,
116        global: &HandlerGlobal<Self::Backend>,
117        cmd: ExportPathMsg,
118    ) -> impl Future<Output = ()> + Send;
119
120    /// Delete the entry, for a delete request or a gc sweep.
121    fn delete(
122        &self,
123        global: &HandlerGlobal<Self::Backend>,
124    ) -> impl Future<Output = std::io::Result<()>> + Send;
125
126    /// Hold off writes and deletes, acknowledging through `ack`, until `release` ends.
127    fn freeze(
128        &self,
129        global: &HandlerGlobal<Self::Backend>,
130        ack: tokio::sync::oneshot::Sender<()>,
131        release: tokio::sync::watch::Receiver<()>,
132    ) -> impl Future<Output = ()> + Send;
133
134    /// Delete the parts a commit stopped referencing, unless the loaded state uses them.
135    fn delete_unreferenced(
136        &self,
137        global: &HandlerGlobal<Self::Backend>,
138        parts: UnreferencedParts,
139    ) -> impl Future<Output = ()> + Send;
140
141    /// Persist the entry when it unloads.
142    fn persist(&self, global: &HandlerGlobal<Self::Backend>) -> impl Future<Output = ()> + Send;
143}

145/// Runs the handlers of backend `B` on the entity manager.
146pub(crate) struct EmParams<B>(PhantomData<fn() -> B>);
147
148impl<B> fmt::Debug for EmParams<B> {
149    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
150        f.write_str("EmParams")
151    }
152}
153
154impl<B: Backend> entity_manager::Params for EmParams<B> {
155    type EntityId = Hash;
156
157    type GlobalState = Arc<HandlerGlobal<B>>;
158
159    type EntityState = B::Handler;
160
161    async fn on_shutdown(
162        state: entity_manager::ActiveEntityState<Self>,
163        cause: entity_manager::ShutdownCause,
164    ) {
165        tracing::trace!("persist {:?} due to {cause:?}", state.id);
166        state.state.persist(&state.global).await;
167    }
168}

170/// A handler as the manager hands it to a task: its hash, the shared state, and the handler.
171pub(crate) type HandlerCtx<B> = ActiveEntityState<EmParams<B>>;

173/// The entity manager is generic over its states; handlers are the store's.
174impl<E: HashHandler> Reset<Hash> for E {
175    fn new(id: &Hash) -> Self {
176        HashHandler::new(id)
177    }
178
179    fn reset(&mut self, id: &Hash) {
180        HashHandler::reset(self, id)
181    }
