37    /// Lists the content protected by temporary tags in all scopes.
38    ///
39    /// An error is the last item in the stream. Premature closure is reported as an error.
40    pub async fn list_temp_tags(
41        &self,
42    ) -> irpc::Result<impl Stream<Item = super::Result<HashAndFormat>>> {
43        self.list_temp_tags_impl(false).await
44    }

46    async fn list_temp_tags_impl(
47        &self,
48        include_dropped: bool,
49    ) -> irpc::Result<impl Stream<Item = super::Result<HashAndFormat>>> {
50        let options = ListTempTagsRequest { include_dropped };
51        trace!("{:?}", options);
52        let mut rx = self.client.server_streaming(options, 32).await?;
53        Ok(Gen::new(|co| async move {
54            let mut rules = TagListRules;
55            loop {
56                let item = match rx.recv().await {
57                    Ok(Some(item)) => item,
58                    Ok(None) => {
59                        co.yield_(Err(super::Error::io(
60                            io::ErrorKind::UnexpectedEof,
61                            "temporary-tag listing ended before completion",
62                        )))
63                        .await;
64                        break;
65                    }
66                    Err(error) => {
67                        co.yield_(Err(error.into())).await;
68                        break;
69                    }
70                };
71                let step = match rules.check(&item) {
72                    Ok(step) => step,
73                    Err(error) => {
74                        co.yield_(Err(error.into())).await;
75                        break;
76                    }
77                };
78                match item {
79                    ListTempTagsItem::Tag(tag) => co.yield_(Ok(tag)).await,
80                    ListTempTagsItem::Done => {}
81                    ListTempTagsItem::Error(error) => co.yield_(Err(error)).await,
82                }
83                if step == Step::Last {
84                    break;
85                }
86            }
87        }))
88    }

90    /// List all tags with options.
91    ///
92    /// This is the most flexible way to list tags. All the other list methods are just convenience
93    /// methods that call this one with the appropriate options.
94    /// An error is the last item in the stream. Premature closure is reported as an error.
95    pub async fn list_with_opts(
96        &self,
97        options: ListOptions,
98    ) -> irpc::Result<impl Stream<Item = super::Result<TagInfo>>> {
99        trace!("{:?}", options);
100        let mut rx = self.client.server_streaming(options, 32).await?;
101        Ok(Gen::new(|co| async move {
102            let mut rules = TagListRules;
103            loop {
104                let item = match rx.recv().await {
105                    Ok(Some(item)) => item,
106                    Ok(None) => {
107                        co.yield_(Err(super::Error::io(
108                            io::ErrorKind::UnexpectedEof,
109                            "tag listing ended before completion",
110                        )))
111                        .await;
112                        break;
113                    }
114                    Err(error) => {
115                        co.yield_(Err(error.into())).await;
116                        break;
117                    }
118                };
119                let step = match rules.check(&item) {
120                    Ok(step) => step,
121                    Err(error) => {
122                        co.yield_(Err(error.into())).await;
123                        break;
124                    }
125                };
126                match item {
127                    ListTagsItem::Tag(tag) => co.yield_(Ok(tag)).await,
128                    ListTagsItem::Done => {}
129                    ListTagsItem::Error(error) => co.yield_(Err(error)).await,
130                }
131                if step == Step::Last {
132                    break;
133                }
134            }
135        }))
136    }

302    #[derive(Clone, Copy)]
303    enum Ending {
304        Done,
305        Error,
306        Drop,
307    }
308
309    // Keep the sender open after a terminal item and send a later item. Clients
310    // must stop at the terminal item without waiting for the sender to close.
311    async fn answer<T: irpc::RpcMessage>(
312        tx: irpc::channel::mpsc::Sender<T>,
313        value: impl Fn() -> T,
314        done: T,
315        error: impl Fn(api::Error) -> T,
316        ending: Ending,
317        empty: bool,
318    ) {
319        if !empty {
320            tx.send(value()).await.unwrap();
321        }
322        let terminal = match ending {
323            Ending::Done => done,
324            Ending::Error => error(api::Error::io(
325                io::ErrorKind::Unsupported,
326                "listing unsupported",
327            )),
328            Ending::Drop => return,
329        };
330        tx.send(terminal).await.unwrap();
331        // A remote client may already have stopped receiving at this point.
332        tx.send(value()).await.ok();
333        tx.closed().await;
334    }
335
336    async fn check_listing<T: std::fmt::Debug + PartialEq>(
337        mut stream: impl Stream<Item = api::Result<T>> + Unpin,
338        expected: T,
339        ending: Ending,
340        empty: bool,
341    ) -> TestResult {
342        if !empty {
343            assert_eq!(stream.next().await.unwrap()?, expected);
344        }
345        match ending {
346            Ending::Done => {}
347            Ending::Error | Ending::Drop => {
348                let api::Error::Io(error) = stream.next().await.unwrap().unwrap_err();
349                let kind = match ending {
350                    Ending::Error => io::ErrorKind::Unsupported,
351                    _ => io::ErrorKind::UnexpectedEof,
352                };
353                assert_eq!(error.kind(), kind);
354            }
355        }
356        assert!(stream.next().await.is_none());
357        Ok(())
358    }

360    async fn listing_endings(remote: bool) -> TestResult {
361        tokio::time::timeout(Duration::from_secs(10), async {
362            for ending in [Ending::Done, Ending::Error, Ending::Drop] {
363                for empty in [false, true] {
364                    let (tx, mut rx) = tokio::sync::mpsc::channel::<Command>(1);
365                    let store = Store::from_client(tx);
366                    let hash = Hash::new("listed");
367                    let responder = tokio::spawn(async move {
368                        let Command::ListTags(cmd) = rx.recv().await.unwrap() else {
369                            panic!("expected ListTags")
370                        };
371                        answer(
372                            cmd.tx,
373                            || ListTagsItem::Tag(TagInfo::new("listed", hash)),
374                            ListTagsItem::Done,
375                            ListTagsItem::Error,
376                            ending,
377                            empty,
378                        )
379                        .await;
380                        let Command::ListTempTags(cmd) = rx.recv().await.unwrap() else {
381                            panic!("expected ListTempTags")
382                        };
383                        answer(
384                            cmd.tx,
385                            || ListTempTagsItem::Tag(HashAndFormat::raw(hash)),
386                            ListTempTagsItem::Done,
387                            ListTempTagsItem::Error,
388                            ending,
389                            empty,
390                        )
391                        .await;
392                    });
393                    #[cfg(feature = "rpc")]
394                    let (store, listener) = if remote {
395                        let addr = "127.0.0.1:0".parse()?;
396                        let (server, cert) = irpc::noq::make_server_endpoint(addr)?;
397                        let client = irpc::noq::make_client_endpoint(addr, &[cert.as_ref()])?;
398                        let address = server.local_addr()?;
399                        let listener = tokio::spawn(store.listen(server));
400                        (Store::connect(client, address), Some(listener))
401                    } else {
402                        (store, None)
403                    };
404                    #[cfg(not(feature = "rpc"))]
405                    assert!(!remote);
406
407                    check_listing(
408                        store.tags().list().await?,
409                        TagInfo::new("listed", hash),
410                        ending,
411                        empty,
412                    )
413                    .await?;
414                    check_listing(
415                        store.tags().list_temp_tags().await?,
416                        HashAndFormat::raw(hash),
417                        ending,
418                        empty,
419                    )
420                    .await?;
421                    responder.await?;
422                    #[cfg(feature = "rpc")]
423                    if let Some(listener) = listener {
424                        listener.abort();
425                    }
426                }
427            }
428            TestResult::Ok(())
429        })
430        .await??;
431        Ok(())
432    }
