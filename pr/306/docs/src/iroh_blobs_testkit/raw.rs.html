1//! Raw protocol requests.
2//!
3//! The client API interprets what the store sends: it stops at the first
4//! error, infers the end of streams and special cases the empty blob. The
5//! helpers here return exactly what the store sent, so cases can check it.

26/// Buffer size for local channels.
27const CAP: usize = 32;

29/// Sends an `ExportBao` request and returns all items until the store ends
30/// the stream.
31pub async fn export_bao(
32    store: &Store,
33    hash: Hash,
34    ranges: ChunkRanges,
35) -> anyhow::Result<Vec<EncodedItem>> {
36    let rx = store
37        .client()
38        .server_streaming(ExportBaoRequest { hash, ranges }, CAP)
39        .await
40        .context("ExportBao request failed")?;
41    collect(rx).await
42}

68/// Returns the first bitfield of an `Observe` stream: the current state.
69pub async fn current_bitfield(store: &Store, hash: Hash) -> anyhow::Result<Bitfield> {

72        .recv()

78/// Sends an `ImportBao` request with `items` and returns the store's answer.
79///
80/// Stops sending early if the store stops reading, e.g. because it rejected
81/// an item.
82pub async fn import_bao(
83    store: &Store,
84    hash: Hash,
85    size: NonZeroU64,
86    items: impl IntoIterator<Item = BaoContentItem>,
87) -> anyhow::Result<()> {
88    let (tx, rx) = store
89        .client()
90        .client_streaming(ImportBaoRequest { hash, size }, CAP)
91        .await
92        .context("ImportBao request failed")?;
93    for item in items {
94        if tx.send(item).await.is_err() {
95            break;
96        }
97    }
98    // Dropping the sender ends the import.
99    drop(tx);
100    let result = rx
101        .await
102        .context("contract: ImportBao answers with a result, but the store dropped the request")?;
103    Ok(result?)
104}

106/// Sends a `DeleteBlobs` request.
107///
108/// Without `force`, the store keeps blobs written since the last
109/// `ClearProtected`.
110pub async fn delete_blobs(store: &Store, hashes: Vec<Hash>, force: bool) -> anyhow::Result<()> {
111    let result = store
112        .client()
113        .rpc(BlobDeleteRequest { hashes, force })

115        .context("DeleteBlobs request failed")?;

138async fn collect<T: irpc::RpcMessage>(mut rx: mpsc::Receiver<T>) -> anyhow::Result<Vec<T>> {
139    let mut items = Vec::new();
140    while let Some(item) = rx.recv().await.context("stream failed")? {
141        items.push(item);
142    }
143    Ok(items)
144}
