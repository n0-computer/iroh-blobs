27/// Checks that the blob in `reference` was added under its hash, is complete
28/// and reads back unchanged.
29pub async fn added(store: &Store, hash: Hash, reference: &Reference) -> anyhow::Result<()> {
30    ensure!(
31        hash == reference.hash(),
32        "contract: a blob is added under the BLAKE3 hash of its content; got {hash}, expected {}",
33        reference.hash()
34    );
35    complete(store, reference).await
36}

38/// Checks that the blob in `reference` is complete and reads back unchanged.
39pub async fn complete(store: &Store, reference: &Reference) -> anyhow::Result<()> {
40    let hash = reference.hash();
41    let size = reference.size();
42    let status = store.status(hash).await?;
43    ensure!(
44        status == BlobStatus::Complete { size },
45        "contract: a complete blob has status Complete {{ size: {size} }}; got {status:?}"
46    );
47    let bitfield = raw::current_bitfield(store, hash).await?;
48    ensure!(
49        bitfield.is_complete() && bitfield.size() == size,
50        "contract: a complete blob has a complete bitfield with size {size}; got {bitfield:?}"
51    );
52    let data = store.get_bytes(hash).await?;
53    ensure!(
54        data == reference.data(),
55        "contract: a complete blob reads back unchanged; got {} bytes that differ",
56        data.len()
57    );
58    Ok(())
59}

61/// Checks that the store has exactly the chunks `present` of the blob in
62/// `reference`, and that they read back unchanged.
63pub async fn has_chunks(
64    store: &Store,
65    reference: &Reference,
66    present: &ChunkRanges,
67) -> anyhow::Result<()> {
68    let hash = reference.hash();
69    let size = reference.size();
70    if *present == reference.all_chunks() {
71        return complete(store, reference).await;
72    }
73
74    let bitfield = raw::current_bitfield(store, hash).await?;
75    let have = reference.clip(&bitfield.ranges);
76    ensure!(
77        have == *present,
78        "contract: the bitfield has exactly the chunks that were imported; got {have:?}, expected {present:?}"
79    );
80    let validated = present.contains(&last_chunk(size));
81    if validated {
82        ensure!(
83            bitfield.size() == size,
84            "contract: once the last chunk is present, the bitfield has the size of the blob ({size}); got {}",
85            bitfield.size()
86        );
87    }
88
89    let status = store.status(hash).await?;
90    match status {
91        BlobStatus::Partial { size: None } if !validated => {}
92        BlobStatus::Partial { size: Some(s) } if s == size => {}
93        _ => bail!(
94            "contract: a partial blob has status Partial with its size{}; got {status:?}",
95            if validated {
96                format!(" ({size}), since the last chunk is present")
97            } else {
98                format!(" ({size}) or no size")
99            }
100        ),
101    }
102
103    let ranges = chunks_to_bytes(present, size);
104    let items = raw::export_ranges(store, hash, ranges.clone()).await?;
105    export_ranges_items(reference, &ranges, items)
106}

108/// Checks the items of an `ExportBao` response for a blob the store has
109/// completely: the size, then exactly the encoding bao-tree produces.
110pub fn export_bao_items(
111    reference: &Reference,
112    ranges: &ChunkRanges,
113    items: Vec<EncodedItem>,
114) -> anyhow::Result<()> {
115    let mut items = items.into_iter();
116    match items.next() {
117        Some(EncodedItem::Size(size)) => ensure!(
118            size == reference.size(),
119            "contract: ExportBao starts with the size of the blob; got {size}, expected {}",
120            reference.size()
121        ),
122        other => bail!("contract: ExportBao starts with the size of the blob; got {other:?}"),
123    }
124    let mut content = Vec::new();
125    let mut end = None;
126    for item in items.by_ref() {
127        match item {
128            EncodedItem::Parent(parent) => content.push(BaoContentItem::Parent(parent)),
129            EncodedItem::Leaf(leaf) => content.push(BaoContentItem::Leaf(leaf)),
130            item => {
131                end = Some(item);
132                break;
133            }
134        }
135    }
136    let content = Encoded::normalize(content);
137    let expected = Encoded::normalize(reference.items(ranges));
138    if let Some(i) =
139        (0..content.len().max(expected.len())).find(|&i| content.get(i) != expected.get(i))
140    {
141        bail!(
142            "contract: ExportBao sends the bao encoding of the requested ranges; part {i} is {:?}, expected {:?}",
143            content.get(i),
144            expected.get(i)
145        );
146    }
147    // Done is allowed but not required: consumers infer the end of the
148    // stream from the size and the ranges.
149    match end {
150        None | Some(EncodedItem::Done) => {}
151        Some(item) => bail!("contract: ExportBao sends nothing after the encoding; got {item:?}"),
152    }
153    ensure!(
154        items.next().is_none(),
155        "contract: ExportBao sends nothing after Done"
156    );
157    Ok(())
158}

160/// Checks the items of an `ExportRanges` response for byte `ranges` the store
161/// has: the size, then the requested bytes in ascending order.
162pub fn export_ranges_items(
163    reference: &Reference,
164    ranges: &RangeSet2<u64>,
165    items: Vec<ExportRangesItem>,
166) -> anyhow::Result<()> {
167    let data = reference.data();
168    let mut items = items.into_iter();
169    match items.next() {
170        Some(ExportRangesItem::Size(size)) => ensure!(
171            size == reference.size(),
172            "contract: ExportRanges starts with the size of the blob; got {size}, expected {}",
173            reference.size()
174        ),
175        other => bail!("contract: ExportRanges starts with the size of the blob; got {other:?}"),
176    }
177    let mut sent = RangeSet2::empty();
178    let mut end = 0;
179    for item in items {
180        let leaf = match item {
181            ExportRangesItem::Data(leaf) => leaf,
182            item => bail!("contract: ExportRanges sends only data after the size; got {item:?}"),
183        };
184        let start = leaf.offset;
185        ensure!(
186            start >= end,
187            "contract: ExportRanges sends data in ascending order without overlap; got offset {start} after data up to {end}"
188        );
189        end = start + leaf.data.len() as u64;
190        ensure!(
191            end <= reference.size() && leaf.data == data[start as usize..end as usize],
192            "contract: ExportRanges sends the content of the blob; data at {start}..{end} differs"
193        );
194        sent |= RangeSet2::from(start..end);
195    }
196    let expected = reference.clip_bytes(ranges);
197    ensure!(
198        sent == expected,
199        "contract: ExportRanges sends exactly the requested bytes within the blob; got {sent:?}, expected {expected:?}"
200    );
201    Ok(())
202}

204/// The chunks a store has after importing `items`.
205pub fn covered_chunks(items: &[BaoContentItem]) -> ChunkRanges {
206    items
207        .iter()
208        .filter_map(|item| match item {
209            BaoContentItem::Leaf(leaf) if !leaf.data.is_empty() => {
210                let start = ChunkNum::full_chunks(leaf.offset);
211                let end = ChunkNum::chunks(leaf.offset + leaf.data.len() as u64);
212                Some(ChunkRanges::from(start..end))
213            }
214            _ => None,
215        })
216        .fold(ChunkRanges::empty(), |acc, range| acc | range)
217}

219fn last_chunk(size: u64) -> ChunkNum {
220    ChunkNum::chunks(size) - 1
221}

223/// A bao encoding with adjacent leaves merged.
224///
225/// How a store splits data into leaves is up to it: the encoding on the wire
226/// is the same.
227#[derive(PartialEq)]
228enum Encoded {
229    /// A parent hash pair. The node of a parent is not part of the contract:
230    /// it is not part of the encoding on the wire, and bao-tree itself does
231    /// not set it for parents within a chunk group.
232    Parent(blake3::Hash, blake3::Hash),
233    Data {
234        offset: u64,
235        data: Vec<u8>,
236    },
237}
238
239impl Encoded {
240    fn normalize(items: Vec<BaoContentItem>) -> Vec<Self> {
241        let mut res = Vec::<Self>::new();
242        for item in items {
243            match item {
244                BaoContentItem::Parent(parent) => {
245                    res.push(Self::Parent(parent.pair.0, parent.pair.1))
246                }
247                BaoContentItem::Leaf(leaf) => match res.last_mut() {
248                    Some(Self::Data { offset, data })
249                        if *offset + data.len() as u64 == leaf.offset =>
250                    {
251                        data.extend_from_slice(&leaf.data)
252                    }
253                    _ => res.push(Self::Data {
254                        offset: leaf.offset,
255                        data: leaf.data.to_vec(),
256                    }),
257                },
258            }
259        }
260        res
261    }
262}
263
264impl fmt::Debug for Encoded {
265    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
266        match self {
267            Self::Parent(left, right) => write!(f, "Parent({left}, {right})"),
268            Self::Data { offset, data } => {
269                // A short content hash, so that data that differs looks different.
270                let hash = blake3::hash(data).to_hex();
271                write!(
272                    f,
273                    "Data({offset}..{}, {})",
274                    offset + data.len() as u64,
275                    &hash[..8]
276                )
277            }
278        }
279    }
280}

282/// How long [`eventually`] waits for a condition.
283const EVENTUALLY: Duration = Duration::from_secs(5);
284
285/// Waits until `condition` holds, for things a store does in the
286/// background, such as releasing temp tags. Fails with `contract` if it
287/// does not hold within a few seconds.
288pub async fn eventually(
289    contract: &str,
290    mut condition: impl AsyncFnMut() -> anyhow::Result<bool>,
291) -> anyhow::Result<()> {
292    let deadline = Instant::now() + EVENTUALLY;
293    while !condition().await? {
294        ensure!(
295            Instant::now() < deadline,
296            "contract: {contract}, but it did not happen within {EVENTUALLY:?}"
297        );
298        tokio::time::sleep(Duration::from_millis(10)).await;
299    }
300    Ok(())
