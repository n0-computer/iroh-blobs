1//! Putting a store into the state a case starts from.
2use anyhow::Context;
3use bao_tree::ChunkRanges;
4use iroh_blobs::{
5    api::{Store, TempTag},
6    Hash,
7};
8
9use super::check;
10use crate::{raw, reference::Reference};
11
12/// Adds the blob and pins it, so a garbage collector keeps it while the
13/// returned temp tag is alive.
14pub async fn add(store: &Store, reference: &Reference) -> anyhow::Result<TempTag> {
15    let tt = store.add_bytes(reference.data().clone()).temp_tag().await?;
16    Ok(tt)
17}
18
19/// Pins a hash, e.g. before importing data for it.
20pub async fn pin(store: &Store, hash: Hash) -> anyhow::Result<TempTag> {
21    Ok(store.tags().temp_tag(hash).await?)
22}
23
24/// Imports the bao encoding of `ranges` of the blob and returns the chunks
25/// the store must have afterwards.
26pub async fn import(
27    store: &Store,
28    reference: &Reference,
29    ranges: &ChunkRanges,
30) -> anyhow::Result<ChunkRanges> {
31    let items = reference.items(ranges);
32    let chunks = check::covered_chunks(&items);
33    raw::import_bao(store, reference.hash(), reference.bao_size(), items)
34        .await
35        .context("contract: importing a valid bao encoding succeeds")?;
36    Ok(chunks)
37}
