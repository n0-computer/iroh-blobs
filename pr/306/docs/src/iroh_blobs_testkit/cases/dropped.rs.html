1//! Requests whose client goes away before they are finished.
2//!
3//! This is normal use: an irpc connection can drop at any time, which closes
4//! all channels of the requests on it. The store must keep working, and must
5//! never claim data it did not get.

18    },

28/// Channel capacity for requests in this module. Small, so the store blocks
29/// on sending when the client goes away.
30const CAP: usize = 1;
31
32/// Each request reads `read` items, then drops the receiver.
33pub fn dropped_exports(h: &Harness) -> anyhow::Result<()> {
34    let inputs = (inputs::non_empty_blob(), 0usize..3);
35    h.check_prop(inputs, async |store, (blob, read)| {
36        let reference = blob.reference();
37        let hash = reference.hash();
38        let _tt = setup::add(&store, &reference).await?;
39        let client = store.client();
40
41        let ranges = ChunkRanges::all();
42        let export_bao = client.server_streaming(ExportBaoRequest { hash, ranges }, CAP);
43        read_and_drop(export_bao.await?, read).await?;
44        let ranges = RangeSet2::all();
45        let export_ranges = client.server_streaming(ExportRangesRequest { hash, ranges }, CAP);
46        read_and_drop(export_ranges.await?, read).await?;
47        let list = client.server_streaming(ListRequest, CAP);
48        read_and_drop(list.await?, read).await?;
49        // An observer of a complete blob gets one item, then waits for
50        // changes that never come.
51        let observe = client.server_streaming(ObserveRequest { hash }, CAP);
52        read_and_drop(observe.await?, read.min(1)).await?;
53
54        check::complete(&store, &reference).await
55    })
56}

58/// The import is cut off after a random number of items.
59pub fn dropped_import_bao(h: &Harness) -> anyhow::Result<()> {
60    let inputs = (inputs::non_empty_blob(), any::<Index>());
61    h.check_prop(inputs, async |store, (blob, cut)| {
62        let reference = blob.reference();
63        let hash = reference.hash();
64        let _tt = setup::pin(&store, hash).await?;
65        let mut items = reference.items(&ChunkRanges::all());
66        items.truncate(cut.index(items.len()));
67        let sent = check::covered_chunks(&items);
68
69        let request = ImportBaoRequest {
70            hash,
71            size: reference.bao_size(),
72        };
73        let (tx, rx) = store.client().client_streaming(request, CAP).await?;
74        for item in items {
75            if tx.send(item).await.is_err() {
76                break;
77            }
78        }
79        drop((tx, rx));
80
81        let bitfield = raw::current_bitfield(&store, hash).await?;
82        let claimed = reference.clip(&bitfield.ranges);
83        ensure!(
84            claimed.is_subset(&sent),
85            "contract: a store only claims chunks it got; claims {claimed:?}, but got only {sent:?}"
86        );
87        if !claimed.is_empty() {
88            let ranges = chunks_to_bytes(&claimed, reference.size());
89            let items = raw::export_ranges(&store, hash, ranges.clone()).await?;
90            check::export_ranges_items(&reference, &ranges, items)?;
91        }
92
93        setup::import(&store, &reference, &ChunkRanges::all()).await?;
94        check::complete(&store, &reference).await
95    })
96}

98/// The connection drops while a byte stream import is in progress.
99pub fn dropped_import_byte_stream(h: &Harness) -> anyhow::Result<()> {
100    let inputs = (
101        inputs::non_empty_blob(),
102        inputs::piece_sizes(),
103        any::<Index>(),
104    );
105    h.check_prop(inputs, async |store, (blob, sizes, cut)| {
106        let reference = blob.reference();
107        let pieces = cut_pieces(reference.data(), &sizes, cut);
108        let prefix = pieces.concat();
109
110        let (tx, rx) = start_byte_stream(&store).await?;
111        drop(rx);
112        send_pieces(tx, pieces).await;
113
114        check_not_added(&store, &prefix).await?;
115        let tt = setup::add(&store, &reference).await?;
116        check::added(&store, tt.hash(), &reference).await
117    })
118}

120/// The client stops sending without `Done`, but still listens for the result.
121pub fn unfinished_import_byte_stream(h: &Harness) -> anyhow::Result<()> {
122    let inputs = (
123        inputs::non_empty_blob(),
124        inputs::piece_sizes(),
125        any::<Index>(),
126    );
127    h.check_prop(inputs, async |store, (blob, sizes, cut)| {
128        let reference = blob.reference();
129        let pieces = cut_pieces(reference.data(), &sizes, cut);
130        let prefix = pieces.concat();
131
132        let (tx, mut rx) = start_byte_stream(&store).await?;
133        // Receive while sending, since the store may block on progress updates.
134        let receive = async {
135            let mut items = Vec::new();
136            while let Some(item) = rx.recv().await? {
137                items.push(item);
138            }
139            anyhow::Ok(items)
140        };
141        let ((), items) = tokio::join!(send_pieces(tx, pieces), receive);
142        let items = items?;
143        if !matches!(items.last(), Some(AddProgressItem::Error(_))) {
144            bail!("contract: an ImportByteStream that ends without Done ends with an error; got {items:?}");
145        }
146
147        check_not_added(&store, &prefix).await
148    })
149}

151pub fn dropped_import_bytes(h: &Harness) -> anyhow::Result<()> {
152    h.check_prop((inputs::blob(), 0usize..3), async |store, (blob, read)| {
153        let reference = blob.reference();
154        let request = ImportBytesRequest {
155            data: reference.data().clone(),
156            format: BlobFormat::Raw,
157            scope: Scope::default(),
158        };
159        read_and_drop(store.client().server_streaming(request, CAP).await?, read).await?;
160
161        let tt = setup::add(&store, &reference).await?;
162        check::added(&store, tt.hash(), &reference).await
163    })
164}

166/// Reads up to `n` items, then drops the receiver.
167async fn read_and_drop<T: irpc::RpcMessage>(
168    mut rx: mpsc::Receiver<T>,
169    n: usize,
170) -> anyhow::Result<()> {
171    for _ in 0..n {
172        if rx.recv().await?.is_none() {
173            break;
174        }
175    }
176    Ok(())
177}

179/// Splits `data` into pieces and keeps only those before `cut`, so the
180/// stream is always incomplete.
181fn cut_pieces(data: &Bytes, sizes: &[usize], cut: Index) -> Vec<Bytes> {
182    let mut pieces = inputs::split(data, sizes);
183    pieces.truncate(cut.index(pieces.len()));
184    pieces
185}

187async fn start_byte_stream(
188    store: &Store,
189) -> anyhow::Result<(
190    mpsc::Sender<ImportByteStreamUpdate>,
191    mpsc::Receiver<AddProgressItem>,
192)> {
193    let request = ImportByteStreamRequest {
194        format: BlobFormat::Raw,
195        scope: Scope::default(),
196    };
197    Ok(store.client().bidi_streaming(request, CAP, CAP).await?)
198}

200/// Sends `pieces` without `Done`, then drops the sender.
201async fn send_pieces(tx: mpsc::Sender<ImportByteStreamUpdate>, pieces: Vec<Bytes>) {
202    for piece in pieces {
203        if tx.send(ImportByteStreamUpdate::Bytes(piece)).await.is_err() {
204            break;
205        }
206    }
207}

209/// Checks that the data of an unfinished byte stream import was not added.
210async fn check_not_added(store: &Store, prefix: &[u8]) -> anyhow::Result<()> {
211    // The store always has the empty blob.
212    if prefix.is_empty() {
213        return Ok(());
214    }
215    let status = store.status(Hash::new(prefix)).await?;
216    ensure!(
217        status == BlobStatus::NotFound,
218        "contract: an ImportByteStream that ends without Done does not add a blob; the {} bytes sent have status {status:?}",
219        prefix.len()
220    );
221    Ok(())
222}
