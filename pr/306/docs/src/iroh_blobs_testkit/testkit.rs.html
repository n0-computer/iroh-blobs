12/// Number of generated inputs per case if `PROPTEST_CASES` is not set.
13const DEFAULT_PROPTEST_CASES: u32 = 32;

15/// Default time limit for one run of a case, including store creation and
16/// shutdown.
17const DEFAULT_TIMEOUT: Duration = Duration::from_secs(30);

19type BoxFuture<T> = Pin<Box<dyn Future<Output = T> + Send>>;
20pub(crate) type Factory = Arc<dyn Fn() -> BoxFuture<anyhow::Result<TestStore>> + Send + Sync>;

22/// All conformance cases.
23pub fn cases() -> &'static [Case] {
24    CASES

27/// A conformance case.
28///
29/// Get all of them with [`cases`], and run one with [`Testkit::run`].
30#[derive(Clone, Copy)]
31pub struct Case {
32    name: &'static str,
33    description: &'static str,
34    run: fn(&Harness) -> anyhow::Result<()>,
35}
36
37impl Case {
38    pub(crate) const fn new(
39        name: &'static str,
40        description: &'static str,
41        run: fn(&Harness) -> anyhow::Result<()>,
42    ) -> Self {
43        Self {
44            name,
45            description,
46            run,
47        }
48    }
49
50    /// The name of the case, e.g. `export_bao`.
51    pub fn name(&self) -> &'static str {
52        self.name
53    }
54
55    /// What the case checks, in one sentence.
56    pub fn description(&self) -> &'static str {
57        self.description
58    }
59}
60
61impl fmt::Debug for Case {
62    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
63        f.debug_struct("Case")
64            .field("name", &self.name)
65            .field("description", &self.description)
66            .finish_non_exhaustive()
67    }
68}

70/// A store under test, created by the factory passed to [`Testkit::new`].
71pub struct TestStore {
72    pub(crate) store: Store,
73    pub(crate) guard: Option<Box<dyn Send>>,
74}
75
76impl TestStore {
77    /// Wraps a store, e.g. a [`MemStore`](iroh_blobs::store::mem::MemStore).
78    pub fn new(store: impl Into<Store>) -> Self {
79        Self {
80            store: store.into(),
81            guard: None,
82        }
83    }
84
85    /// Keeps `guard` alive until the store has been shut down.
86    ///
87    /// Use this for resources the store needs, such as a temp dir or a server
88    /// task.
89    pub fn with_guard(mut self, guard: impl Send + 'static) -> Self {
90        self.guard = Some(Box::new(guard));
91        self
92    }
93}
94
95impl fmt::Debug for TestStore {
96    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
97        f.debug_struct("TestStore")
98            .field("store", &self.store)
99            .finish_non_exhaustive()
100    }
101}

103/// Runs conformance cases against stores from a factory.
104#[derive(Clone)]
105pub struct Testkit {
106    factory: Factory,
107    settings: Settings,
108    skip: BTreeMap<&'static str, String>,
109}
110
111impl fmt::Debug for Testkit {
112    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
113        f.debug_struct("Testkit")
114            .field("settings", &self.settings)
115            .field("skip", &self.skip)
116            .finish_non_exhaustive()
117    }
118}
119
120impl Testkit {
121    /// Creates a testkit for stores created by `factory`.
122    ///
123    /// The factory is called for every generated input and must return a
124    /// fresh, empty store each time.
125    pub fn new<F, Fut>(factory: F) -> Self
126    where
127        F: Fn() -> Fut + Send + Sync + 'static,
128        Fut: Future<Output = anyhow::Result<TestStore>> + Send + 'static,
129    {
130        let proptest_cases = std::env::var("PROPTEST_CASES")
131            .ok()
132            .and_then(|s| s.parse().ok())
133            .unwrap_or(DEFAULT_PROPTEST_CASES);
134        Self {
135            factory: Arc::new(move || Box::pin(factory())),
136            settings: Settings {
137                proptest_cases,
138                timeout: DEFAULT_TIMEOUT,
139            },
140            skip: BTreeMap::new(),
141        }
142    }
143
144    /// Sets the number of generated inputs per case.
145    ///
146    /// Defaults to the `PROPTEST_CASES` environment variable, or 32.
147    pub fn proptest_cases(mut self, cases: u32) -> Self {
148        self.settings.proptest_cases = cases;
149        self
150    }
151
152    /// Sets the time limit for one run of a case, including store creation
153    /// and shutdown. Defaults to 30 seconds.
154    ///
155    /// A run that takes longer fails, since a store must answer every
156    /// request.
157    pub fn timeout(mut self, timeout: Duration) -> Self {
158        self.settings.timeout = timeout;
159        self
160    }
161
162    /// Marks a case as skipped, e.g. because it tests a known bug.
163    ///
164    /// [`Testkit::main`] reports skipped cases as ignored, so they still run
165    /// with `--include-ignored`.
166    ///
167    /// # Panics
168    ///
169    /// If there is no case named `case`.
170    pub fn skip(mut self, case: &str, reason: impl Into<String>) -> Self {
171        let case = cases()
172            .iter()
173            .find(|c| c.name == case)
174            .unwrap_or_else(|| panic!("Testkit::skip: there is no case named {case:?}"));
175        self.skip.insert(case.name, reason.into());
176        self
177    }
178
179    /// The reason a case is skipped, if it is.
180    pub fn skip_reason(&self, case: &Case) -> Option<&str> {
181        self.skip.get(case.name).map(String::as_str)
182    }
183
184    /// Runs one case, whether or not it is skipped.
185    ///
186    /// On failure, the error states the violated contract and the minimal
187    /// failing input.
188    pub fn run(&self, case: &Case) -> anyhow::Result<()> {
189        let harness = Harness::new(self.factory.clone(), self.settings);
190        (case.run)(&harness)
191    }
192
193    /// One [`Trial`] per case, for running with [libtest-mimic].
194    ///
195    /// Skipped cases are marked as ignored.
196    ///
197    /// [libtest-mimic]: https://docs.rs/libtest-mimic
198    pub fn trials(&self) -> Vec<Trial> {
199        cases()
200            .iter()
201            .map(|case| {
202                let testkit = self.clone();
203                let case = *case;
204                Trial::test(case.name, move || {
205                    testkit
206                        .run(&case)
207                        .map_err(|cause| Failed::from(format!("{cause:#}")))
208                })
209                .with_ignored_flag(self.skip.contains_key(case.name))
210            })
211            .collect()
212    }
213
214    /// Runs all cases like `cargo test` does, then exits.
215    ///
216    /// Use this as the `main` of a test target with `harness = false`. It
217    /// supports the usual test arguments, such as a name filter, `--list` and
218    /// `--include-ignored`.
219    pub fn main(&self) -> ! {
220        let args = Arguments::from_args();
221        if !args.list {
222            for (name, reason) in &self.skip {
223                eprintln!("skipping {name}: {reason}");
224            }
225        }
226        libtest_mimic::run(&args, self.trials()).exit()
227    }
