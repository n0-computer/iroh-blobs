1//! Running one case: store lifecycle, timeouts and property test setup.
2use std::time::Duration;
3
4use anyhow::{anyhow, bail, Context};
5use iroh_blobs::api::Store;
6use proptest::{
7    strategy::Strategy,
8    test_runner::{Config as ProptestConfig, TestCaseError, TestRunner},
9};
10
11use crate::testkit::{Factory, TestStore};
12
13/// Settings for running cases.
14#[derive(Debug, Clone, Copy)]
15pub(crate) struct Settings {
16    /// Number of generated inputs per property.
17    pub proptest_cases: u32,
18    /// Time limit for one run, including store creation and shutdown.
19    pub timeout: Duration,
20}
21
22/// Runs the checks of one case, each run against a fresh store.
23pub(crate) struct Harness {
24    runtime: tokio::runtime::Runtime,
25    factory: Factory,
26    settings: Settings,
27}
28
29impl Harness {
30    pub fn new(factory: Factory, settings: Settings) -> Self {
31        let runtime = tokio::runtime::Builder::new_multi_thread()
32            .worker_threads(2)
33            .enable_all()
34            .build()
35            .expect("failed to create tokio runtime");
36        Self {
37            runtime,
38            factory,
39            settings,
40        }
41    }
42
43    /// Runs `test` against a fresh store for every input generated by
44    /// `strategy`. On failure, the error contains the minimal failing input.
45    pub fn check_prop<S>(
46        &self,
47        strategy: S,
48        test: impl AsyncFn(Store, S::Value) -> anyhow::Result<()>,
49    ) -> anyhow::Result<()>
50    where
51        S: Strategy,
52    {
53        let config = ProptestConfig {
54            cases: self.settings.proptest_cases,
55            // Failures would be persisted relative to this crate's sources,
56            // which are read only when used as a dependency.
57            failure_persistence: None,
58            ..ProptestConfig::default()
59        };
60        let mut runner = TestRunner::new(config);
61        runner
62            .run(&strategy, |input| {
63                self.runtime
64                    .block_on(self.with_store(async |store| test(store, input).await))
65                    .map_err(|cause| TestCaseError::fail(format!("{cause:#}")))
66            })
67            .map_err(|cause| anyhow!("{cause}"))
68    }
69
70    /// Creates a store, runs `test` against it and shuts it down.
71    async fn with_store(
72        &self,
73        test: impl AsyncFnOnce(Store) -> anyhow::Result<()>,
74    ) -> anyhow::Result<()> {
75        let timeout = self.settings.timeout;
76        let TestStore { store, guard } = tokio::time::timeout(timeout, (self.factory)())
77            .await
78            .map_err(|_| anyhow!("store factory timed out after {timeout:?}"))?
79            .context("store factory failed")?;
80        let result = match tokio::time::timeout(timeout, test(store.clone())).await {
81            Ok(result) => result,
82            Err(_) => Err(anyhow!(
83                "timed out after {timeout:?}: a store must answer every request and end every stream"
84            )),
85        };
86        // Shut down even if the test failed, so the store releases its resources.
87        let shutdown = tokio::time::timeout(timeout, store.shutdown()).await;
88        drop(guard);
89        result?;
90        match shutdown {
91            Ok(Ok(())) => Ok(()),
92            Ok(Err(cause)) => bail!("shutdown failed: {cause}"),
93            Err(_) => bail!("shutdown timed out after {timeout:?}"),
94        }
95    }
96}
