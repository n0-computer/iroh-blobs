1//! Expected values for a blob, computed with bao-tree.
2use std::{io::Cursor, num::NonZeroU64};
3
4use bao_tree::{
5    io::{
6        outboard::PreOrderMemOutboard,
7        sync::{encode_ranges_validated, DecodeResponseIter},
8        BaoContentItem,
9    },
10    ChunkNum, ChunkRanges,
11};
12use bytes::Bytes;
13use iroh_blobs::{store::IROH_BLOCK_SIZE, Hash};
14use range_collections::{range_set::RangeSetRange, RangeSet2};
15
16/// A blob together with its outboard.
17pub struct Reference {
18    data: Bytes,
19    outboard: PreOrderMemOutboard,
20}
21
22impl Reference {
23    pub fn new(data: Bytes) -> Self {
24        let outboard = PreOrderMemOutboard::create(&data, IROH_BLOCK_SIZE);
25        Self { data, outboard }
26    }
27
28    pub fn data(&self) -> &Bytes {
29        &self.data
30    }
31
32    pub fn hash(&self) -> Hash {
33        self.outboard.root.into()
34    }
35
36    pub fn size(&self) -> u64 {
37        self.data.len() as u64
38    }
39
40    /// The size for a bao import, which needs a non-empty blob.
41    pub fn bao_size(&self) -> NonZeroU64 {
42        NonZeroU64::new(self.size()).expect("bao import needs a non-empty blob")
43    }
44
45    /// All chunks of the blob.
46    pub fn all_chunks(&self) -> ChunkRanges {
47        ChunkRanges::from(..ChunkNum::chunks(self.size()))
48    }
49
50    /// `ranges`, limited to the chunks of the blob.
51    pub fn clip(&self, ranges: &ChunkRanges) -> ChunkRanges {
52        ranges & &self.all_chunks()
53    }
54
55    /// The bao encoding of `ranges`, as a sequence of items.
56    pub fn items(&self, ranges: &ChunkRanges) -> Vec<BaoContentItem> {
57        let mut encoded = Vec::new();
58        encode_ranges_validated(&self.data[..], &self.outboard, ranges, &mut encoded)
59            .expect("encoding from complete data and outboard can not fail");
60        DecodeResponseIter::new(
61            self.outboard.root,
62            self.outboard.tree,
63            Cursor::new(encoded),
64            ranges,
65        )
66        .collect::<Result<_, _>>()
67        .expect("a valid encoding always decodes")
68    }
69
70    /// `ranges`, limited to the bytes of the blob.
71    pub fn clip_bytes(&self, ranges: &RangeSet2<u64>) -> RangeSet2<u64> {
72        ranges & &RangeSet2::from(0..self.size())
73    }
74}
75
76/// The byte ranges covered by `ranges`, limited to `size`.
77pub fn chunks_to_bytes(ranges: &ChunkRanges, size: u64) -> RangeSet2<u64> {
78    let mut bytes = RangeSet2::empty();
79    for range in ranges.iter() {
80        let (start, end) = match range {
81            RangeSetRange::Range(range) => (range.start.to_bytes(), range.end.to_bytes()),
82            RangeSetRange::RangeFrom(range) => (range.start.to_bytes(), size),
83        };
84        let end = end.min(size);
85        if start < end {
86            bytes |= RangeSet2::from(start..end);
87        }
88    }
89    bytes
90}
