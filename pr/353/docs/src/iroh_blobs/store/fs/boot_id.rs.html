1//! Identification of the current boot session of the machine.
2//!
3//! The store records the boot ID while it is open. If it finds a recorded ID
4//! from a different boot, the machine went down while the store was open, so
5//! an OS crash or power loss may have lost data that was never synced.
6
7/// Returns an ID that is the same for the whole boot session and new after every boot.
8///
9/// Returns `None` where the platform has no such ID, or it cannot be read.
10pub(super) fn current() -> Option<String> {
11    let id = platform::current()?;
12    let id = id.trim();
13    (!id.is_empty()).then(|| id.to_owned())
14}

16#[cfg(any(target_os = "linux", target_os = "android"))]
17mod platform {
18    /// The kernel creates a new random UUID on every boot.
19    pub(super) fn current() -> Option<String> {
20        std::fs::read_to_string("/proc/sys/kernel/random/boot_id").ok()
21    }
22}

24#[cfg(any(target_os = "macos", target_os = "ios"))]
25mod platform {
26    use std::ffi::CStr;
27
28    /// The kernel creates a new UUID for every boot session.
29    pub(super) fn current() -> Option<String> {
30        let mut buf = [0u8; 64];
31        let mut len = buf.len();
32        // SAFETY: the name is a nul terminated string, `buf` is a writable
33        // buffer of `len` bytes, and `len` carries its size in and the length
34        // of the value out. No new value is set.
35        let res = unsafe {
36            libc::sysctlbyname(
37                c"kern.bootsessionuuid".as_ptr(),
38                buf.as_mut_ptr().cast(),
39                &mut len,
40                std::ptr::null_mut(),
41                0,
42            )
43        };
44        if res != 0 {
45            return None;
46        }
47        let value = CStr::from_bytes_until_nul(buf.get(..len)?).ok()?;
48        value.to_str().ok().map(str::to_owned)
49    }
50}

52#[cfg(windows)]
53mod platform {
54    use windows_sys::Win32::{
55        Foundation::ERROR_SUCCESS,
56        System::Registry::{RegGetValueW, HKEY_LOCAL_MACHINE, RRF_RT_REG_DWORD},
57    };
58
59    /// The prefetcher counts boots in the registry, once per boot.
60    pub(super) fn current() -> Option<String> {
61        let key = wide(
62            r"SYSTEM\CurrentControlSet\Control\Session Manager\Memory Management\PrefetchParameters",
63        );
64        let name = wide("BootId");
65        let mut value = 0u32;
66        let mut size = std::mem::size_of::<u32>() as u32;
67        // SAFETY: key and name are nul terminated wide strings, and `value` is
68        // a writable DWORD whose size `size` carries in and out.
69        let res = unsafe {
70            RegGetValueW(
71                HKEY_LOCAL_MACHINE,
72                key.as_ptr(),
73                name.as_ptr(),
74                RRF_RT_REG_DWORD,
75                std::ptr::null_mut(),
76                std::ptr::addr_of_mut!(value).cast(),
77                &mut size,
78            )
79        };
80        (res == ERROR_SUCCESS).then(|| value.to_string())
81    }
82
83    /// Encodes `s` as a nul terminated UTF-16 string.
84    fn wide(s: &str) -> Vec<u16> {
85        s.encode_utf16().chain(std::iter::once(0)).collect()
86    }
87}
