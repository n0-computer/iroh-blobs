9/// Options for directories used by the file store.
10#[derive(Debug, Clone)]
11pub struct PathOptions {
12    /// Path to the directory where data and outboard files are stored.
13    pub data_path: PathBuf,
14    /// Path to the directory where temp files are stored.
15    /// This *must* be on the same device as `data_path`, since we need to
16    /// atomically move temp files into place.
17    pub temp_path: PathBuf,
18}
19
20impl PathOptions {
21    pub fn new(root: &Path) -> Self {
22        Self {
23            data_path: root.join("data"),
24            temp_path: root.join("temp"),
25        }
26    }
27
28    pub fn data_path(&self, hash: &Hash) -> PathBuf {
29        self.data_path.join(format!("{}.data", hash.to_hex()))
30    }
31
32    pub fn outboard_path(&self, hash: &Hash) -> PathBuf {
33        self.data_path.join(format!("{}.obao4", hash.to_hex()))
34    }
35
36    pub fn sizes_path(&self, hash: &Hash) -> PathBuf {
37        self.data_path.join(format!("{}.sizes4", hash.to_hex()))
38    }
39
40    pub fn bitfield_path(&self, hash: &Hash) -> PathBuf {
41        self.data_path.join(format!("{}.bitfield", hash.to_hex()))
42    }
43
44    pub fn temp_file_name(&self) -> PathBuf {
45        self.temp_path.join(temp_name())
46    }
47}

49/// Options for inlining small complete data or outboards.
50#[derive(Debug, Clone)]
51pub struct InlineOptions {
52    /// Maximum data size to inline.
53    pub max_data_inlined: u64,
54    /// Maximum outboard size to inline.
55    pub max_outboard_inlined: u64,
56}
57
58impl InlineOptions {
59    /// Do not inline anything, ever.
60    pub const NO_INLINE: Self = Self {
61        max_data_inlined: 0,
62        max_outboard_inlined: 0,
63    };
64    /// Always inline everything
65    pub const ALWAYS_INLINE: Self = Self {
66        max_data_inlined: u64::MAX,
67        max_outboard_inlined: u64::MAX,
68    };
69}
70
71impl Default for InlineOptions {
72    fn default() -> Self {
73        Self {
74            max_data_inlined: 1024 * 16,
75            max_outboard_inlined: 1024 * 16,
76        }
77    }
78}

80/// Options for transaction batching.

104/// What kind of crash the file store stays consistent through.
105///
106/// This is about consistency, not durability. After a crash the store holds
107/// no entry that claims data it does not have, but recent writes can be lost
108/// in every mode, even acknowledged ones. A write is only durable after
109/// [`Store::sync_db`](crate::api::Store::sync_db) or a clean shutdown, and
110/// only against the kind of crash the chosen level covers.
111#[derive(Debug, Clone, Copy, Default, PartialEq, Eq)]
112#[non_exhaustive]
113pub enum CrashConsistency {
114    /// Consistent after a process crash; recovers from an OS crash or power loss on the next open.
115    ///
116    /// No data files are synced, so an OS crash or power loss can leave entries
117    /// that claim data that never reached the disk. The store records the boot
118    /// id while it is open, and if it finds one from another boot session on
119    /// open, every entry with files checks them again when it is next loaded.
120    /// Until then it may report less data than it has, never more.
121    ///
122    /// On Linux and Android, a clean shutdown syncs the filesystem before it
123    /// forgets the boot id. Elsewhere, an OS crash within seconds after a clean
124    /// shutdown, before the OS wrote out its cache, goes unnoticed. Where there
125    /// is no boot id, every unclean shutdown counts as a possible OS crash.
126    #[default]
127    ProcessCrash,
128    /// Consistent after an OS crash or power loss too, at the cost of fsyncs.
129    ///
130    /// Data files and the data directory are synced before the metadata that
131    /// depends on them is recorded, for imports as well as downloads. That
132    /// costs a few fsyncs for every blob above the inline threshold, which
133    /// slows down imports and downloads of many small blobs considerably.
134    PowerLoss,
135}

137/// Options for the file store.
138#[derive(Debug, Clone)]
139pub struct Options {
140    /// Path options.
141    pub path: PathOptions,
142    /// Inline storage options.
143    pub inline: InlineOptions,
144    /// Transaction batching options.
145    pub batch: BatchOptions,
146    /// Gc configuration.
147    pub gc: Option<GcConfig>,
148    /// What kind of crash the store stays consistent through.
149    pub crash_consistency: CrashConsistency,
150}
151
152impl Options {
153    /// Create new optinos with the given root path and everything else default.
154    pub fn new(root: &Path) -> Self {
155        Self {
156            path: PathOptions::new(root),
157            inline: InlineOptions::default(),
158            batch: BatchOptions::default(),
159            gc: None,
160            crash_consistency: CrashConsistency::default(),
161        }
162    }
163
164    /// True if files must reach the disk before metadata that depends on them is recorded.
165    pub(crate) fn syncs_files(&self) -> bool {
166        self.crash_consistency == CrashConsistency::PowerLoss
167    }
168
169    // check if the data will be inlined, based on the size of the data
170    pub fn is_inlined_data(&self, data_size: u64) -> bool {
171        data_size <= self.inline.max_data_inlined
172    }
173
174    // check if the outboard will be inlined, based on the size of the *outboard*
175    pub fn is_inlined_outboard(&self, outboard_size: u64) -> bool {
176        outboard_size <= self.inline.max_outboard_inlined
177    }
178
179    // check if both the data and outboard will be inlined, based on the size of the data
180    pub fn is_inlined_all(&self, data_size: u64) -> bool {
181        let outboard_size = raw_outboard_size(data_size);
182        self.is_inlined_data(data_size) && self.is_inlined_outboard(outboard_size)
183    }
184}
