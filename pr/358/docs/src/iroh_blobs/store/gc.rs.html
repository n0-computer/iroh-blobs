37/// Compute the set of live hashes
38pub(super) async fn gc_mark_task(
39    store: &Store,
40    live: &mut HashSet<Hash>,
41    roots: &[HashAndFormat],
42    co: &Co<GcMarkEvent>,
43) -> crate::api::Result<()> {
44    macro_rules! trace {
45        ($($arg:tt)*) => {
46            co.yield_(GcMarkEvent::CustomDebug(format!($($arg)*))).await;
47        };
48    }
49    macro_rules! warn {
50        ($($arg:tt)*) => {
51            co.yield_(GcMarkEvent::CustomWarning(format!($($arg)*), None)).await;
52        };
53    }
54    // Every root is live, and HashSeq roots also have their children traversed.
55    // Traversal does not depend on `live`, which may already contain a root
56    // because of a Raw tag on the same hash or the protect callback.
57    let mut seqs = HashSet::new();
58    let mut add_root = |HashAndFormat { hash, format }: HashAndFormat| {
59        live.insert(hash);
60        if !format.is_raw() {
61            seqs.insert(hash);
62        }
63    };
64    // Entry freezes and tag suspension cover this exact root snapshot until
65    // the GC guard is dropped. Never re-list roots during the session.
66    for &root in roots {
67        add_root(root);
68    }
69    for hash in seqs {
70        let mut stream = store.export_bao(hash, ChunkRanges::all()).hashes();
71        while let Some(hash) = stream.next().await {
72            match hash {
73                Ok(hash) => {
74                    live.insert(hash);
75                }
76                Err(e) => {
77                    warn!("error while traversing hashseq: {e:?}");
78                }
79            }
80        }
81    }
82    trace!("gc mark done. found {} live blobs", live.len());
83    Ok(())
84}

86async fn gc_sweep_task(
87    store: &Store,
88    run: &crate::api::blobs::GcRun,
89    live: &HashSet<Hash>,
90    co: &Co<GcSweepEvent>,
91) -> crate::api::Result<()> {
92    let mut count = 0;
93    let mut batch = Vec::new();
94    for &hash in &run.snapshot.candidates {
95        if !live.contains(&hash) {
96            batch.push(hash);
97            count += 1;
98        }
99        if batch.len() >= 100 {
100            run.delete(store, batch.clone()).await?;
101            batch.clear();
102        }
103    }
104    if !batch.is_empty() {
105        run.delete(store, batch).await?;
106    }
107    store.sync_db().await?;
108    co.yield_(GcSweepEvent::CustomDebug(format!(
109        "requested deletion of {count} blobs"
110    )))
111    .await;
112    Ok(())
113}

115fn gc_mark<'a>(
116    store: &'a Store,
117    live: &'a mut HashSet<Hash>,
118    roots: &'a [HashAndFormat],
119) -> impl Stream<Item = GcMarkEvent> + 'a {
120    Gen::new(|co| async move {
121        if let Err(e) = gc_mark_task(store, live, roots, &co).await {
122            co.yield_(GcMarkEvent::Error(e)).await;
123        }
124    })
125}

127fn gc_sweep<'a>(

139/// Configuration for garbage collection.
140///
141/// To protect blobs during long-running writes without pausing the GC
142/// schedule, use [`crate::api::blobs::Batch::temp_tag`].
143#[derive(derive_more::Debug, Clone)]
144pub struct GcConfig {
145    /// Interval in which to run garbage collection.
146    pub interval: Duration,
147    /// Optional callback to manually add protected blobs.
148    ///
149    /// The callback is called after marking, before sweeping. It gets a `&mut HashSet<Hash>`
150    /// and returns a future that returns [`ProtectOutcome`]. It must not await tag
151    /// mutations, imports that publish a temp tag, or writes to protected HashSeqs:
152    /// those are suspended during GC.
153    /// All hashes that are added to the
154    /// [`HashSet`] will be protected from garbage collection during this run.
155    ///
156    /// In normal operation, return [`ProtectOutcome::Continue`] from the callback. If you return
157    /// [`ProtectOutcome::Abort`], the garbage collection run will be aborted.Use this if your
158    /// source of hashes to protect returned an error, and thus garbage collection should be skipped
159    /// completely to not unintentionally delete blobs that should be protected.
160    #[debug("ProtectCallback")]
161    pub add_protected: Option<ProtectCb>,
162}
163
164/// Returned from [`ProtectCb`].
165///
166/// See [`GcConfig::add_protected] for details.
167#[derive(Debug)]
168pub enum ProtectOutcome {
169    /// Continue with the garbage collection run.
170    Continue,
171    /// Abort the garbage collection run.
172    Abort,
173}

175/// The type of the garbage collection callback.
176///
177/// See [`GcConfig::add_protected] for details.
178pub type ProtectCb = Arc<
179    dyn for<'a> Fn(
180            &'a mut HashSet<Hash>,
181        )
182            -> Pin<Box<dyn std::future::Future<Output = ProtectOutcome> + Send + Sync + 'a>>
183        + Send
184        + Sync
185        + 'static,
186>;

188/// Runs a single mark-and-sweep garbage collection cycle.
189///
190/// Entering GC captures candidates and roots, freezes HashSeq data, and defers
191/// all tag mutations (including imports that publish a temp tag). Available
192/// chunks of incomplete roots are traversed without waiting for missing data.
193/// Blobs absent from the candidate snapshot cannot be swept this round.
194///
195/// The optional `protect` callback ([`GcConfig::add_protected`]) runs after
196/// marking, before sweep. It must only add hashes to `live`, and must not wait
197/// for a tag mutation or a write to a frozen HashSeq. External writers must
198/// acquire store protection before publishing references, and establish/check
199/// content after acquisition if it was previously unprotected.
200///
201/// Completion, errors and cancellation release the session. The store waits
202/// for accepted sweep deletes before resuming deferred tag publication, and
203/// rejects requests arriving after their session ends.
204pub async fn gc_run_once(
205    store: &Store,
206    live: &mut HashSet<Hash>,
207    protect: Option<&ProtectCb>,
208) -> crate::api::Result<()> {
209    debug!(externally_protected = live.len(), "gc: start");
210    // Enter returns only after tag mutations are suspended and every captured
211    // HashSeq root is frozen. Dropping the guard exits on success/error/cancel.
212    let run = store.gc_run().await?;
213    {
214        let mut stream = gc_mark(store, live, &run.snapshot.roots);
215        while let Some(ev) = stream.next().await {
216            match ev {
217                GcMarkEvent::CustomDebug(msg) => {
218                    debug!("{}", msg);
219                }
220                GcMarkEvent::CustomWarning(msg, err) => {
221                    warn!("{}: {:?}", msg, err);
222                }
223                GcMarkEvent::Error(err) => {
224                    error!("error during gc mark: {:?}", err);
225                    return Err(err);
226                }
227            }
228        }
229    }
230    // Capture the external protect set after the mark phase, just before the sweep.
231    if let Some(protect) = protect {
232        match (protect)(live).await {
233            ProtectOutcome::Continue => {}
234            ProtectOutcome::Abort => {
235                info!("abort gc run: protect callback indicated abort");
236                return Ok(());
237            }
238        }
239    }
240    debug!(total_protected = live.len(), "gc: sweep");
241    {
242        let mut stream = gc_sweep(store, &run, live);
243        while let Some(ev) = stream.next().await {
244            match ev {
245                GcSweepEvent::CustomDebug(msg) => {
246                    debug!("{}", msg);
247                }
248                GcSweepEvent::CustomWarning(msg, err) => {
249                    warn!("{}: {:?}", msg, err);
250                }
251                GcSweepEvent::Error(err) => {
252                    error!("error during gc sweep: {:?}", err);
253                    return Err(err);
254                }
255            }
256        }
257    }
258    debug!("gc: done");
259
260    Ok(())
261}

263/// Runs garbage collection in a loop until the store is gone.
264///
265/// This holds only a weak sender, so it does not keep the store alive. Once all
266/// handles to the store are dropped, the loop ends.
267pub async fn run_gc(store: WeakSender<Command>, config: GcConfig) {
268    debug!("gc enabled with interval {:?}", config.interval);
269    let mut live = HashSet::new();
270    loop {
271        live.clear();
272        n0_future::time::sleep(config.interval).await;
273        let Some(sender) = store.upgrade() else {
274            debug!("gc: store dropped, stopping");
275            break;
276        };
277        let store = Store::from_client(sender);
278        if let Err(e) = gc_run_once(&store, &mut live, config.add_protected.as_ref()).await {
279            error!("error during gc run: {e}");
280            break;
281        }
282    }
283}

453    /// New blobs are deferred to the next candidate snapshot, even without a
454    /// tag. Existing candidates still need current or retained temp protection.
455    async fn gc_import_after_mark_survives(store: &Store) -> TestResult<()> {
456        use bao_tree::ChunkNum;
457
458        use crate::{api::blobs::BlobStatus, store::util::tests::create_n0_bao};
459        let kept = store.add_slice(vec![0u8; 100000]).temp_tag().await?;
460        let dropped = store.add_slice(vec![2u8; 100000]).temp_tag().await?;
461        let dropped_hash = dropped.hash();
462        let old = store.add_slice("old untagged blob").temp_tag().await?;
463        let old_hash = old.hash();
464        drop(old);
465        store.sync_db().await?;
466        let run = store.gc_run().await?;
467        let candidates = &run.snapshot.candidates;
468        assert!(candidates.contains(&kept.hash()));
469        assert!(candidates.contains(&dropped_hash));
470        assert!(candidates.contains(&old_hash));
471        // Model a mark that missed these roots. Temp-tag retention protects
472        // candidates even when a tag is dropped before sweep.
473        let live = HashSet::new();
474        drop(dropped);
475        // Written after candidate collection, without any tag.
476        let data = vec![1u8; 100000];
477        let ranges = ChunkRanges::from(..ChunkNum(16));
478        let (untagged, bao) = create_n0_bao(&data, &ranges)?;
479        store.import_bao_bytes(untagged, ranges, bao).await?;
480        let mut stream = gc_sweep(store, &run, &live);
481        while let Some(ev) = stream.next().await {
482            if let GcSweepEvent::Error(err) = ev {
483                return Err(err.into());
484            }
485        }
486        assert!(store.has(kept.hash()).await?, "a temp tag protects");
487        assert!(
488            store.has(dropped_hash).await?,
489            "a dropped tag protects this round"
490        );
491        assert!(
492            !store.has(old_hash).await?,
493            "old unprotected candidates are swept"
494        );
495        assert!(matches!(
496            store.status(untagged).await?,
497            BlobStatus::Partial { .. }
498        ));
499        drop(stream);
500        drop(run);
501
502        // Deferral lasts only this round; there is no retained write-protection set.
503        gc_run_once(store, &mut HashSet::new(), None).await?;
504        assert_eq!(store.status(untagged).await?, BlobStatus::NotFound);
505        assert!(store.has(kept.hash()).await?);
506        Ok(())
507    }

534    /// A partial root remains readable but cannot gain references; a complete
535    /// root cannot be deleted/recreated. Unrelated writes/deletes still proceed.
536    async fn gc_freezes_hashseqs(store: &Store) -> TestResult {
537        use crate::{api::blobs::DeleteOptions, store::util::tests::create_n0_bao};
538        use bao_tree::ChunkNum;
539        let child = store.add_slice("known child").temp_tag().await?;
540        let hash = child.hash();
541        let late = store.add_slice("unpublished child").temp_tag().await?;
542        let late_hash = late.hash();
543        drop(late);
544        let data: bytes::Bytes = std::iter::repeat_n(hash, 512)
545            .chain(std::iter::repeat_n(late_hash, 512))
546            .chain(std::iter::repeat_n(Hash::new("unavailable"), 512))
547            .collect::<HashSeq>()
548            .into();
549        let first = ChunkRanges::from(..ChunkNum(16));
550        let (root, bao) = create_n0_bao(&data, &first)?;
551        let root_tag = store.tags().temp_tag(HashAndFormat::hash_seq(root)).await?;
552        store
553            .import_bao_bytes(root, first.clone(), bao.clone())
554            .await?;
555        let complete_data: bytes::Bytes = [hash].into_iter().collect::<HashSeq>().into();
556        let complete = store
557            .add_bytes_with_opts(AddBytesOptions {
558                data: complete_data.clone(),
559                format: BlobFormat::HashSeq,
560            })
561            .temp_tag()
562            .await?;
563        let complete_hash = complete.hash();
564        let garbage = store.add_slice("garbage").temp_tag().await?;
565        let garbage_hash = garbage.hash();
566        drop(garbage);
567        store.sync_db().await?;
568
569        let run = store.gc_run().await?;
570        assert!(run.delete(store, vec![complete_hash]).await.is_err());
571        assert!(run.snapshot.candidates.contains(&late_hash));
572        assert!(run.snapshot.roots.contains(&root_tag.hash_and_format()));
573        let writer_store = store.clone();
574        let mut writer = tokio::spawn(async move {
575            let range = ChunkRanges::from(ChunkNum(16)..ChunkNum(32));
576            let (_, bao) = create_n0_bao(&data, &range)?;
577            writer_store.import_bao_bytes(root, range, bao).await?;
578            TestResult::Ok(())
579        });
580        let delete_store = store.clone();
581        let mut delete = tokio::spawn(async move {
582            delete_store
583                .blobs()
584                .delete_with_opts(DeleteOptions {
585                    hashes: vec![complete_hash],
586                    force: true,
587                    gc_id: None,
588                })
589                .await
590        });
591        assert_pending(&mut writer).await;
592        assert_pending(&mut delete).await;
593        assert_eq!(store.export_bao(root, first).bao_to_vec().await?, bao);
594        assert_eq!(store.get_bytes(complete_hash).await?, complete_data);
595        let mut observed = store.observe(root).stream().await?;
596        assert!(observed.next().await.is_some());
597
598        // A raw download needs no tag operation and is outside the candidates.
599        let new_data = b"new raw download";
600        let (new_hash, new_bao) = create_n0_bao(new_data, &ChunkRanges::all())?;
601        store
602            .import_bao_bytes(new_hash, ChunkRanges::all(), new_bao)
603            .await?;
604        let mut live = HashSet::new();
605        let mut mark = gc_mark(store, &mut live, &run.snapshot.roots);
606        while let Some(event) = mark.next().await {
607            if let GcMarkEvent::Error(err) = event {
608                return Err(err.into());
609            }
610        }
611        drop(mark);
612        assert!(live.contains(&hash));
613        assert!(!live.contains(&late_hash));
614        let mut sweep = gc_sweep(store, &run, &live);
615        while let Some(event) = sweep.next().await {
616            if let GcSweepEvent::Error(err) = event {
617                return Err(err.into());
618            }
619        }
620        drop(sweep);
621        assert!(!store.has(garbage_hash).await?);
622        assert!(store.has(new_hash).await?);
623        assert_pending(&mut writer).await;
624        assert_pending(&mut delete).await;
625        drop(run);
626        writer.await??;
627        delete.await??;
628        assert!(!store.has(complete_hash).await?);
629        // The previously unknown reference is only published after sweep.
630        let mut hashes = store
631            .export_bao(root, ChunkRanges::from(..ChunkNum(32)))
632            .hashes();
633        let mut seen = HashSet::new();
634        while let Some(hash) = hashes.next().await {
635            seen.insert(hash?);
636        }
637        assert!(seen.contains(&late_hash));
638        drop(hashes);
639        drop(observed);
640        drop(root_tag);
641        drop(complete);
642        drop(child);
643        Ok(())
644    }

646    async fn gc_suspends_tag_mutations(store: &Store) -> TestResult {
647        let a = store.add_slice("a").temp_tag().await?;
648        let b = store.add_slice("b").temp_tag().await?;
649        let ah = a.hash();
650        let bh = b.hash();
651        for name in ["set", "rename", "delete"] {
652            store.tags().set(name, ah).await?;
653        }
654        let run = store.gc_run().await?;
655        let mut changes = Vec::new();
656        for kind in 0..5 {
657            let store = store.clone();
658            changes.push(tokio::spawn(async move {
659                match kind {
660                    0 => store.tags().set("set", bh).await?,
661                    1 => {
662                        store.tags().create(bh).await?;
663                    }
664                    2 => store.tags().rename("rename", "renamed").await?,
665                    3 => {
666                        store.tags().delete("delete").await?;
667                    }
668                    4 => {
669                        store
670                            .tags()
671                            .set("hashseq", HashAndFormat::hash_seq(bh))
672                            .await?
673                    }
674                    _ => unreachable!(),
675                }
676                TestResult::Ok(())
677            }));
678        }
679        let mut temps = Vec::new();
680        for value in [HashAndFormat::raw(bh), HashAndFormat::hash_seq(bh)] {
681            let store = store.clone();
682            temps.push(tokio::spawn(
683                async move { store.tags().temp_tag(value).await },
684            ));
685        }
686        let import_store = store.clone();
687        let mut import =
688            tokio::spawn(async move { import_store.add_slice("new import").temp_tag().await });
689        for task in &mut changes {
690            assert_pending(task).await;
691        }
692        for task in &mut temps {
693            assert_pending(task).await;
694        }
695        assert_pending(&mut import).await;
696        // Reads work, and replacements/deletes have not taken effect.
697        let mut tags = store.tags().list().await?;
698        let mut before = Vec::new();
699        while let Some(tag) = tags.next().await {
700            before.push(tag?);
701        }
702        assert_eq!(before.len(), 3);
703        assert!(before.iter().all(|tag| tag.hash == ah));
704        drop(tags);
705        // The root captured at entry remains protected even when its tag drops.
706        drop(a);
707        assert!(run.snapshot.roots.contains(&HashAndFormat::raw(ah)));
708        drop(run);
709        for task in changes {
710            task.await??;
711        }
712        let mut held = Vec::new();
713        for task in temps {
714            held.push(task.await??);
715        }
716        held.push(import.await??);
717        let mut tags = store.tags().list().await?;
718        let mut names = std::collections::BTreeSet::new();
719        while let Some(tag) = tags.next().await {
720            names.insert(tag?.name);
721        }
722        assert!(names.contains(&crate::api::Tag::from("renamed")));
723        assert!(!names.contains(&crate::api::Tag::from("delete")));
724        Ok(())
725    }

758    /// A delete delayed in transit must be rejected if its GC session ended,
759    /// including when another session has already started.
760    async fn gc_cancel_during_delete(store: &Store) -> TestResult {
761        use tokio::sync::oneshot;
762        let garbage = store
763            .add_slice("cancelled-sweep candidate")
764            .temp_tag()
765            .await?;
766        let hash = garbage.hash();
767        drop(garbage);
768        store.sync_db().await?;
769        let inner = store.client().as_local().expect("local store");
770        let (tx, mut rx) = tokio::sync::mpsc::channel::<Command>(32);
771        let (queued, queued_rx) = oneshot::channel();
772        let (release, released) = oneshot::channel();
773        let responder = tokio::spawn(async move {
774            let mut pause = Some((queued, released));
775            while let Some(cmd) = rx.recv().await {
776                if matches!(cmd, Command::DeleteBlobs(_)) {
777                    if let Some((queued, released)) = pause.take() {
778                        queued.send(()).unwrap();
779                        released.await.unwrap();
780                    }
781                }
782                if inner.send_raw(cmd).await.is_err() {
783                    break;
784                }
785            }
786        });
787        let proxy = Store::from_client(tx);
788        let gc_store = proxy.clone();
789        let gc =
790            tokio::spawn(async move { gc_run_once(&gc_store, &mut HashSet::new(), None).await });
791        queued_rx.await?;
792        gc.abort();
793        assert!(gc.await.unwrap_err().is_cancelled());
794        let tag_store = store.clone();
795        let tag = tokio::spawn(async move { tag_store.tags().temp_tag(hash).await });
796        let _tag = tag.await??;
797        let next = store.gc_run().await?;
798        release.send(()).unwrap();
799        drop(proxy);
800        responder.await?;
801        assert!(store.has(hash).await?, "a stale sweep deleted the blob");
802        drop(next);
803        Ok(())
804    }

1014    /// The callback reports external references while concurrent tag publication
1015    /// waits. Its snapshot and the store snapshot together protect old content.
1016    async fn gc_protect_race(store: &Store) -> TestResult {
1017        use tokio::sync::{mpsc, oneshot};
1018        let tt = store.add_slice("external").temp_tag().await?;
1019        let hash = tt.hash();
1020        drop(tt);
1021        let (entered_tx, mut entered_rx) = mpsc::channel::<oneshot::Sender<()>>(1);
1022        let cb: ProtectCb = Arc::new(move |live| {
1023            live.insert(hash);
1024            let entered_tx = entered_tx.clone();
1025            Box::pin(async move {
1026                let (release, released) = oneshot::channel();
1027                entered_tx.send(release).await.unwrap();
1028                released.await.unwrap();
1029                ProtectOutcome::Continue
1030            })
1031        });
1032        let gc_store = store.clone();
1033        let gc =
1034            tokio::spawn(
1035                async move { gc_run_once(&gc_store, &mut HashSet::new(), Some(&cb)).await },
1036            );
1037        let release = entered_rx.recv().await.unwrap();
1038        let writer_store = store.clone();
1039        let mut writer = tokio::spawn(async move { writer_store.tags().temp_tag(hash).await });
1040        assert_pending(&mut writer).await;
1041        assert!(store.has(hash).await?);
1042        release.send(()).unwrap();
1043        gc.await??;
1044        let _tag = writer.await??;
1045        assert!(store.has(hash).await?);
1046        Ok(())
1047    }
