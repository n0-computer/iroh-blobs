98    /// Creates a new batch scope.
99    ///
100    /// A batch holds open a set of temporary tags that protect blobs from
101    /// garbage collection for as long as the batch is alive. When the batch is
102    /// dropped, all its temporary tags are released and the protected blobs
103    /// become eligible for collection again.
104    ///
105    /// Use a batch whenever a long-running operation needs to protect a blob
106    /// in the middle of a write, e.g. when downloading a large blob that takes
107    /// longer than the GC interval. Without a temp tag, GC may collect the
108    /// partially written data file before the operation completes.
109    ///
110    /// See [`Batch::temp_tag`] to pin a specific hash inside the batch scope.
111    pub async fn batch(&self) -> irpc::Result<Batch<'_>> {
112        let msg = BatchRequest;
113        trace!("{msg:?}");
114        let (tx, rx) = self.client.client_streaming(msg, 32).await?;
115        let scope = rx.await?;
116
117        Ok(Batch {
118            scope,
119            blobs: self,
120            _tx: tx,
121        })
122    }

156    /// Delete a blob.
157    ///
158    /// GC uses this after excluding live roots and their children from its
159    /// candidate snapshot. Unless forced, deletion also respects current and
160    /// GC-retained temp tags. Named tags are accounted for by GC's mark phase.
161    ///
162    /// Users should rely only on garbage collection for blob deletion.
163    pub(crate) async fn delete_with_opts(&self, mut options: DeleteOptions) -> RequestResult<()> {
164        trace!("{options:?}");
165        // the empty blob is always present, so it is never deleted
166        options.hashes.retain(|hash| *hash != Hash::EMPTY);
167        self.client.rpc(options).await??;
168        Ok(())
169    }

281    pub async fn add_stream(
282        &self,
283        data: impl Stream<Item = io::Result<Bytes>> + Send + 'static,
284    ) -> AddProgress<'_> {
285        let inner = ImportByteStreamRequest {
286            format: crate::BlobFormat::Raw,
287            scope: Scope::default(),
288        };
289        let client = self.client.clone();
290        let stream = Gen::new(|co| async move {
291            let (sender, mut receiver) = match client.bidi_streaming(inner, 32, 32).await {
292                Ok(x) => x,
293                Err(cause) => {
294                    co.yield_(AddProgressItem::Error(cause.into())).await;
295                    return;
296                }
297            };
298            let recv = async {
299                loop {
300                    match receiver.recv().await {
301                        Ok(Some(item)) => co.yield_(item).await,
302                        Err(cause) => {
303                            co.yield_(AddProgressItem::Error(cause.into())).await;
304                            break;
305                        }
306                        Ok(None) => break,
307                    }
308                }
309            };
310            // Moving the sender in means it is dropped when the source fails,
311            // which makes the store abort the import instead of waiting forever.
312            let send = async move {
313                tokio::pin!(data);
314                while let Some(item) = data.next().await {
315                    let bytes = item.map_err(AddProgressItem::Error)?;
316                    if sender
317                        .send(ImportByteStreamUpdate::Bytes(bytes))
318                        .await
319                        .is_err()
320                    {
321                        // the store side is gone, recv will report why
322                        return Ok(());
323                    }
324                }
325                sender.send(ImportByteStreamUpdate::Done).await.ok();
326                Ok(())
327            };
328            // If send succeeds its branch is disabled and we keep receiving.
329            // On a source error, report it rather than the store's generic
330            // "byte stream ended unexpectedly".
331            tokio::select! {
332                Err(item) = send => co.yield_(item).await,
333                _ = recv => {}
334            }
335        });
336        AddProgress::new(self, stream)
337    }

395    pub async fn export_chunk(
396        &self,
397        hash: impl Into<Hash>,
398        offset: u64,
399    ) -> super::ExportBaoResult<Leaf> {
400        let base = ChunkNum::full_chunks(offset);
401        let ranges = ChunkRanges::from(base..base + 1);
402        let mut stream = self.export_bao(hash, ranges).stream();
403        while let Some(item) = stream.next().await {
404            match item {
405                EncodedItem::Leaf(leaf) => return Ok(leaf),
406                EncodedItem::Parent(_) => {}
407                EncodedItem::Size(_) => {}
408                EncodedItem::Done => break,
409                EncodedItem::Error(cause) => return Err(cause.into()),
410            }
411        }
412        Err(io::Error::other("unexpected end of stream").into())
413    }

480    #[cfg_attr(feature = "hide-proto-docs", doc(hidden))]
481    pub async fn import_bao_reader<R: crate::util::RecvStream>(
482        &self,
483        hash: Hash,
484        ranges: ChunkRanges,
485        mut reader: R,
486    ) -> RequestResult<R> {
487        let mut size = [0; 8];
488        reader.recv_exact(&mut size).await?;
489        let size = u64::from_le_bytes(size);
490        let Some(size) = NonZeroU64::new(size) else {
491            return if hash == Hash::EMPTY {
492                Ok(reader)
493            } else {
494                Err(io::Error::other("invalid size for hash").into())
495            };
496        };
497        let tree = BaoTree::new(size.get(), IROH_BLOCK_SIZE);
498        let mut decoder = ResponseDecoder::new(
499            hash.into(),
500            ranges,
501            tree,
502            RecvStreamAsyncStreamReader::new(reader),
503        );
504        let options = ImportBaoOptions { hash, size };
505        let handle = self.import_bao_with_opts(options, 32).await?;
506        let driver = async move {
507            let reader = loop {
508                match decoder.next().await {
509                    ResponseDecoderNext::More((rest, item)) => {
510                        handle.tx.send(item?).await?;
511                        decoder = rest;
512                    }
513                    ResponseDecoderNext::Done(reader) => break reader,
514                };
515            };
516            drop(handle.tx);
517            io::Result::Ok(reader)
518        };
519        let fut = async move { handle.rx.await.map_err(io::Error::other)? };
520        let (reader, res) = tokio::join!(driver, fut);
521        res?;
522        Ok(reader?.into_inner())
523    }

627/// A scoped lifetime that protects blobs from garbage collection.
628///
629/// All temporary tags created through a `Batch` are kept alive until the batch
630/// is dropped, at which point they are released atomically and the protected
631/// blobs become eligible for GC again.
632///
633/// A typical use case is protecting a blob during a long-running write
634/// (or download) where GC could otherwise collect the partially-written data.
635///
636/// # Example
637///
638/// ```rust
639/// use iroh_blobs::{store::mem::MemStore, Hash};
640///
641/// # async fn example() -> n0_error::Result<()> {
642/// let store = MemStore::new();
643///
644/// // Hash is known upfront (e.g. from a download ticket).
645/// let hash = Hash::new(b"example content");
646///
647/// let batch = store.blobs().batch().await?;
648/// let _guard = batch.temp_tag(hash).await?;
649/// // GC will not collect `hash` for as long as `_guard` and `batch` are alive.
650///
651/// // ... perform long-running write or network download here ...
652///
653/// drop(_guard);
654/// // Any un-upgraded blobs from the batch are now eligible for garbage collection
655/// # Ok(())
656/// # }
657/// ```
658pub struct Batch<'a> {
659    scope: Scope,
660    blobs: &'a Blobs,
661    _tx: mpsc::Sender<BatchResponse>,
662}

726/// A progress handle for an import operation.
727///
728/// Internally this is a stream of [`AddProgressItem`] items. Working with this
729/// stream directly can be inconvenient, so this struct provides some convenience
730/// methods to work with the result.
731///
732/// It also implements [`IntoFuture`], so you can await it to get the [`TagInfo`] that
733/// contains the hash of the added content. Awaiting it is the same as calling
734/// [`AddProgress::with_tag`]: it creates a persistent tag with an automatically
735/// generated name, which protects the content from garbage collection until you
736/// delete the tag.
737///
738/// To protect the content only temporarily, use [`AddProgress::temp_tag`] instead.
739/// To choose the tag name, use [`AddProgress::with_named_tag`].
740///
741/// If you want access to the stream, you can use the [`AddProgress::stream`] method.
742pub struct AddProgress<'a> {
743    blobs: &'a Blobs,
744    inner: stream::Boxed<AddProgressItem>,
745}
746
747impl<'a> IntoFuture for AddProgress<'a> {
748    type Output = RequestResult<TagInfo>;
749
750    type IntoFuture = Pin<Box<dyn Future<Output = Self::Output> + Send + 'a>>;
751
752    fn into_future(self) -> Self::IntoFuture {
753        Box::pin(self.with_tag())
754    }
755}

757impl<'a> AddProgress<'a> {
758    fn new(blobs: &'a Blobs, stream: impl Stream<Item = AddProgressItem> + Send + 'static) -> Self {
759        let mut stream = Box::pin(stream);
760        let checked = Gen::new(|co| async move {
761            let mut rules = AddProgressRules::default();
762            while let Some(item) = stream.next().await {
763                let step = match rules.check(&item) {
764                    Ok(step) => step,
765                    Err(cause) => {
766                        co.yield_(AddProgressItem::Error(cause)).await;
767                        return;
768                    }
769                };
770                co.yield_(item).await;
771                if step == Step::Last {
772                    return;
773                }
774            }
775        });
776        Self {
777            blobs,
778            inner: Box::pin(checked),
779        }
780    }

782    /// Waits for the import to finish and returns a [`TempTag`] for the content.
783    ///
784    /// The content is protected from garbage collection only as long as the
785    /// temp tag is alive. No persistent tag is created.
786    pub async fn temp_tag(self) -> RequestResult<TempTag> {
787        let mut stream = self.inner;
788        while let Some(item) = stream.next().await {
789            match item {
790                AddProgressItem::Done(tt) => return Ok(tt),
791                AddProgressItem::Error(e) => return Err(e.into()),
792                _ => {}
793            }
794        }
795        Err(io::Error::other("unexpected end of stream").into())
796    }
797
798    /// Waits for the import to finish and sets the persistent tag `name` to the content.
799    ///
800    /// Any previous value of that tag is replaced. The content is protected from
801    /// garbage collection until the tag is deleted or set to something else.
802    pub async fn with_named_tag(self, name: impl AsRef<[u8]>) -> RequestResult<HashAndFormat> {
803        let blobs = self.blobs.clone();
804        let tt = self.temp_tag().await?;
805        let haf = tt.hash_and_format();
806        let tags = Tags::ref_from_sender(&blobs.client);
807        tags.set(name, haf).await?;
808        drop(tt);
809        Ok(haf)
810    }
811
812    /// Waits for the import to finish and creates a persistent tag with a generated name.
813    ///
814    /// The name looks like `auto-2026-01-01T00:00:00.000Z`.
815    ///
816    /// This is what awaiting [`AddProgress`] does. The content is protected from
817    /// garbage collection until the tag is deleted, so if you don't need a
818    /// persistent tag, use [`AddProgress::temp_tag`] instead.
819    pub async fn with_tag(self) -> RequestResult<TagInfo> {
820        let blobs = self.blobs.clone();
821        let tt = self.temp_tag().await?;
822        let hash = tt.hash();
823        let format = tt.format();
824        let tags = Tags::ref_from_sender(&blobs.client);
825        let name = tags.create(tt.hash_and_format()).await?;
826        drop(tt);
827        Ok(TagInfo { name, hash, format })
828    }

832    }

1157        let mut rx = self.inner.await?;
1158        let mut rules = ExportRangesRules::new(self.ranges.clone());
1159        let mut size = None;
1160        let mut data = BTreeMap::new();
1161        loop {

1163            let last = rules.check(&item)? == Step::Last;
1164            match item {
1165                ExportRangesItem::Size(s) => size = Some(s),
1166                ExportRangesItem::Data(leaf) => {
1167                    data.insert(leaf.offset, leaf.data);
1168                }
1169                ExportRangesItem::Error(cause) => return Err(cause.into()),

1171            if last {

1178                RangeSetRange::RangeFrom(range) => (*range.start, u64::MAX),
1179                RangeSetRange::Range(range) => (*range.start, *range.end),
1180            };
1181            for (offset, data) in data.iter() {
1182                let cstart = *offset;
1183                let cend = *offset + (data.len() as u64);
1184                if cstart >= end || cend <= start {
1185                    continue;
1186                }
1187                let start = start.max(cstart);
1188                let end = end.min(cend);
1189                let data = &data[(start - cstart) as usize..(end - cstart) as usize];
1190                res.extend_from_slice(data);
1191            }
1192        }
1193        Ok((size, res))
1194    }

1334            let item = rx.recv().await?.ok_or_else(export_bao_ended_early)?;
1335            let last = rules.check(&item)? == Step::Last;
1336            match item {
1337                EncodedItem::Size(size) => {
1338                    target.write(&size.to_le_bytes()).await?;
1339                }
1340                EncodedItem::Parent(parent) => {
1341                    let mut data = vec![0u8; 64];
1342                    data[..32].copy_from_slice(parent.pair.0.as_bytes());
1343                    data[32..].copy_from_slice(parent.pair.1.as_bytes());
1344                    target.write(&data).await?;
1345                }
1346                EncodedItem::Leaf(leaf) => {
1347                    target.write_bytes(leaf.data).await?;
1348                }
1349                EncodedItem::Done => {}
1350                EncodedItem::Error(cause) => return Err(cause.into()),
1351            }
1352            if last {
1353                break;
1354            }
1355        }
1356        Ok(())
1357    }

1359    /// Write noq variant that also feeds a progress writer.
1360    pub(crate) async fn write_with_progress<W: crate::util::SendStream>(
1361        self,
1362        writer: &mut W,
1363        progress: &mut impl WriteProgress,
1364        hash: &Hash,
1365        index: u64,
1366    ) -> super::ExportBaoResult<()> {
1367        let mut rx = self.inner.await?;
1368        let mut rules = ExportBaoRules::new(&self.ranges);
1369        loop {
1370            let item = rx.recv().await?.ok_or_else(export_bao_ended_early)?;
1371            let last = rules.check(&item)? == Step::Last;
1372            match item {
1373                EncodedItem::Size(size) => {
1374                    progress.send_transfer_started(index, hash, size).await;
1375                    writer.send(&size.to_le_bytes()).await?;
1376                    progress.log_other_write(8);
1377                }
1378                EncodedItem::Parent(parent) => {
1379                    let mut data = [0u8; 64];
1380                    data[..32].copy_from_slice(parent.pair.0.as_bytes());
1381                    data[32..].copy_from_slice(parent.pair.1.as_bytes());
1382                    writer.send(&data).await?;
1383                    progress.log_other_write(64);
1384                }
1385                EncodedItem::Leaf(leaf) => {
1386                    let len = leaf.data.len();
1387                    writer.send_bytes(leaf.data).await?;
1388                    progress
1389                        .notify_payload_write(index, leaf.offset, len)
1390                        .await?;
1391                }
1392                EncodedItem::Done => {}
1393                EncodedItem::Error(cause) => return Err(cause.into()),
1394            }
1395            if last {
1396                break;
1397            }
1398        }
1399        Ok(())
1400    }

1402    pub fn into_byte_stream(self) -> impl Stream<Item = super::Result<Bytes>> {
1403        self.stream().filter_map(|item| match item {
1404            EncodedItem::Size(size) => {
1405                let size = size.to_le_bytes().to_vec().into();
1406                Some(Ok(size))
1407            }
1408            EncodedItem::Parent(parent) => {
1409                let mut data = vec![0u8; 64];
1410                data[..32].copy_from_slice(parent.pair.0.as_bytes());
1411                data[32..].copy_from_slice(parent.pair.1.as_bytes());
1412                Some(Ok(data.into()))
1413            }
1414            EncodedItem::Leaf(leaf) => Some(Ok(leaf.data)),
1415            EncodedItem::Done => None,
1416            EncodedItem::Error(cause) => Some(Err(cause.into())),
1417        })
1418    }

1420    pub fn stream(self) -> impl Stream<Item = EncodedItem> {
1421        Gen::new(|co| async move {
1422            let mut rx = match self.inner.await {
1423                Ok(rx) => rx,
1424                Err(cause) => {
1425                    co.yield_(EncodedItem::Error(io::Error::other(cause).into()))
1426                        .await;
1427                    return;
1428                }
1429            };
1430            let mut rules = ExportBaoRules::new(&self.ranges);
1431            let item = loop {
1432                match rx.recv().await {
1433                    Ok(Some(item)) => {
1434                        let step = match rules.check(&item) {
1435                            Ok(step) => step,
1436                            Err(cause) => break EncodedItem::Error(cause.into()),
1437                        };
1438                        co.yield_(item).await;
1439                        if step == Step::Last {
1440                            return;
1441                        }
1442                    }
1443                    Ok(None) => break EncodedItem::Error(export_bao_ended_early_io().into()),
1444                    Err(cause) => break EncodedItem::Error(io::Error::other(cause).into()),
1445                }
1446            };
1447            co.yield_(item).await;
1448        })
1449    }
1450}
